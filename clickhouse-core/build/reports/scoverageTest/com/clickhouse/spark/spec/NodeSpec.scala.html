30 case class NodeSpec(
31   @JsonIgnore private val _host: String,
32   @JsonIgnore private val _http_port: Option[Int] = None,
33   @JsonIgnore private val _tcp_port: Option[Int] = None,
34   @JsonProperty("protocol") protocol: ClickHouseProtocol = HTTP,
35   @JsonProperty("username") username: String = "default",
36   @JsonProperty("password") password: String = "",
37   @JsonProperty("database") database: String = "default",
38   @JsonProperty("options") options: util.Map[String, String] = Collections.emptyMap()
39 ) extends Nodes with ToJson with Serializable {
40   @JsonProperty("host") def host: String = findHost(_host)
41   @JsonProperty("http_port") def http_port: Option[Int] = findPort(_http_port)
42   @JsonProperty("tcp_port") def tcp_port: Option[Int] = findPort(_tcp_port)
43 
44   private def findHost(source: String): String =
45     if (isTesting) {
46       // workaround for testcontainers docker compose network mechanism
47       sys.props.get(s"${PREFIX}_HOST_$source").getOrElse(source)
48     } else source
49 
50   private def findPort(source: Option[Int]): Option[Int] =
51     if (isTesting) {
52       // workaround for testcontainers docker compose network mechanism
53       source.map(p => sys.props.get(s"${PREFIX}_HOST_${_host}_PORT_$p").map(_.toInt).getOrElse(p))
54     } else source
55 
56   def port: Int = protocol match {
57     case HTTP => http_port.get
58     case TCP => tcp_port.get
59     case unsupported => throw new IllegalArgumentException(s"Unsupported protocol: $unsupported")
60   }
61 
62   override def toString: String = s"[$protocol]$username@$host:$port}/$database"
63 
64   @JsonIgnore @transient override lazy val nodes: Array[NodeSpec] = Array(this)
65 }
