20 sealed trait TableEngineSpec extends Serializable {
21   def engine_clause: String
22   def engine: String
23   def args: List[Expr] = List.empty
24   def sorting_key: TupleExpr = TupleExpr(List.empty)
25   def primary_key: TupleExpr = TupleExpr(List.empty)
26   def partition_key: TupleExpr = TupleExpr(List.empty)
27   def sampling_key: TupleExpr = TupleExpr(List.empty)
28   def ttl: Option[String] = None // don't care about it now
29   def settings: Map[String, String]
30   def is_distributed: Boolean = false
31   def is_replicated: Boolean = false
32   def order_by_expr: List[OrderExpr] = sorting_key.exprList.map(OrderExpr(_))
33 }
