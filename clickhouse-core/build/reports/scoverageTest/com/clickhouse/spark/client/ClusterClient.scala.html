29 class ClusterClient(cluster: ClusterSpec) extends AutoCloseable with Logging {
30 
31   @transient lazy val cache = new ConcurrentHashMap[(Int, Int), NodeClient]
32 
33   def node(shard: Option[Int] = None, replica: Option[Int] = None): NodeClient = {
34     val (_shard, _replica) = (shard, replica) match {
35       case (Some(s), Some(r)) => (s, r)
36       case (Some(s), None) =>
37         val shardSpec = cluster.shards.filter(_.num == s).head
38         val replicaSpec = shuffle(shardSpec.replicas.toSeq).head
39         (s, replicaSpec.num)
40       case (None, None) =>
41         val shardSpec = shuffle(cluster.shards.toSeq).head
42         val replicaSpec = shuffle(shardSpec.replicas.toSeq).head
43         (shardSpec.num, replicaSpec.num)
44       case _ => throw CHClientException(
45           s"Invalid shard[${shard.orNull}] replica[${replica.orNull}] of cluster ${cluster.name}"
46         )
47     }
48 
49     cache.computeIfAbsent(
50       (_shard, _replica),
51       { case (s, r) =>
52         val shardSpec = cluster.shards.find(_.num == s).get
53         val replicaSpec = shardSpec.replicas.find(_.num == r).get
54         val nodeSpec = replicaSpec.node
55         log.info(s"Create client to $nodeSpec, shard $s replica $r")
56         new NodeClient(nodeSpec)
57       }
58     )
59   }
60 
61   override def close(): Unit = cache.asScala.values.foreach(_.close())
62 }
