63   override def visitEngineClause(ctx: EngineClauseContext): TableEngineSpec = {
64     val engineExpr = source(ctx.engineExpr)
65     val engine = source(ctx.engineExpr.identifierOrNull)
66 
67     val engineArgs: Seq[Expr] =
68       Option(ctx.engineExpr.columnExprList)
69         .map(_.columnsExpr.asScala)
70         .getOrElse(List.empty)
71         .map(visitColumnsExpr)
72         .toSeq
73 
74     val orderByOpt = listToOption(ctx.orderByClause)
75       .map(visitOrderByClause)
76       .map(orderByList => tupleIfNeeded(orderByList.map(_.expr)))
77     val pkOpt = listToOption(ctx.primaryKeyClause)
78       .map(_.columnExpr)
79       .map(visitColumnExpr)
80     val partOpt = listToOption(ctx.partitionByClause)
81       .map(_.columnExpr)
82       .map(visitColumnExpr)
83     val sampleByOpt = listToOption(ctx.sampleByClause)
84       .map(_.columnExpr)
85       .map(visitColumnExpr)
86     val ttlOpt = listToOption(ctx.ttlClause).map(source) // we don't care about ttl now
87     val settings = listToOption(ctx.settingsClause)
88       .map(visitSettingsClause)
89       .getOrElse(Map.empty)
90 
91     engine match {
92       case eg: String if "MergeTree" equalsIgnoreCase eg =>
93         MergeTreeEngineSpec(
94           engine_clause = engineExpr,
95           _sorting_key = tupleIfNeeded(orderByOpt.toList),
96           _primary_key = tupleIfNeeded(pkOpt.toList),
97           _partition_key = tupleIfNeeded(partOpt.toList),
98           _sampling_key = tupleIfNeeded(sampleByOpt.toList),
99           _ttl = ttlOpt,
100           _settings = settings
101         )
102       case eg: String if "ReplacingMergeTree" equalsIgnoreCase eg =>
103         ReplacingMergeTreeEngineSpec(
104           engine_clause = engineExpr,
105           version_column = engineArgs.lift(0).map(_.asInstanceOf[FieldRef]),
106           is_deleted_column = engineArgs.lift(1).map(_.asInstanceOf[FieldRef]),
107           _sorting_key = tupleIfNeeded(orderByOpt.toList),
108           _primary_key = tupleIfNeeded(pkOpt.toList),
109           _partition_key = tupleIfNeeded(partOpt.toList),
110           _sampling_key = tupleIfNeeded(sampleByOpt.toList),
111           _ttl = ttlOpt,
112           _settings = settings
113         )
114       case eg: String if "ReplicatedMergeTree" equalsIgnoreCase eg =>
115         ReplicatedMergeTreeEngineSpec(
116           engine_clause = engineExpr,
117           zk_path = engineArgs.head.asInstanceOf[StringLiteral].value,
118           replica_name = engineArgs(1).asInstanceOf[StringLiteral].value,
119           _sorting_key = tupleIfNeeded(orderByOpt.toList),
120           _primary_key = tupleIfNeeded(pkOpt.toList),
121           _partition_key = tupleIfNeeded(partOpt.toList),
122           _sampling_key = tupleIfNeeded(sampleByOpt.toList),
123           _ttl = ttlOpt,
124           _settings = settings
125         )
126       case eg: String if "ReplicatedReplacingMergeTree" equalsIgnoreCase eg =>
127         ReplicatedReplacingMergeTreeEngineSpec(
128           engine_clause = engineExpr,
129           zk_path = engineArgs.head.asInstanceOf[StringLiteral].value,
130           replica_name = engineArgs(1).asInstanceOf[StringLiteral].value,
131           version_column = engineArgs.lift(2).map(_.asInstanceOf[FieldRef]),
132           is_deleted_column = engineArgs.lift(3).map(_.asInstanceOf[FieldRef]),
133           _sorting_key = tupleIfNeeded(orderByOpt.toList),
134           _primary_key = tupleIfNeeded(pkOpt.toList),
135           _partition_key = tupleIfNeeded(partOpt.toList),
136           _sampling_key = tupleIfNeeded(sampleByOpt.toList),
137           _ttl = ttlOpt,
138           _settings = settings
139         )
140       case eg: String if "Distributed" equalsIgnoreCase eg =>
141         DistributedEngineSpec(
142           engine_clause = engineExpr,
143           cluster = engineArgs.head.asInstanceOf[StringLiteral].value,
144           local_db = engineArgs(1).asInstanceOf[StringLiteral].value,
145           local_table = engineArgs(2).asInstanceOf[StringLiteral].value,
146           sharding_key = engineArgs.drop(3).headOption,
147           _settings = settings
148         )
149       case _ => UnknownTableEngineSpec(engineExpr)
150     }
151   }

156   def visitColumnExpr(ctx: ColumnExprContext): Expr = ctx match {
157     case fieldCtx: ColumnExprIdentifierContext => visitColumnExprIdentifier(fieldCtx)
158     case literalCtx: ColumnExprLiteralContext => visitColumnExprLiteral(literalCtx)
159     case funcCtx: ColumnExprFunctionContext => visitColumnExprFunction(funcCtx)
160     case singleColCtx: ColumnExprParensContext => visitColumnExpr(singleColCtx.columnExpr)
161     case tupleCtx: ColumnExprTupleContext => visitColumnExprTuple(tupleCtx)
162     case precedence1Ctx: ColumnExprPrecedence1Context => visitColumnExprPrecedence1(precedence1Ctx)
163     case precedence2Ctx: ColumnExprPrecedence2Context => visitColumnExprPrecedence2(precedence2Ctx)
164     case other: ColumnExprContext => throw new IllegalArgumentException(
165         s"Unsupported ColumnExpr: [${other.getClass.getSimpleName}] ${other.getText}"
166       )
167   }
168 
169   override def visitColumnExprIdentifier(ctx: ColumnExprIdentifierContext): FieldRef =
170     FieldRef(source(ctx.columnIdentifier))
171 
172   override def visitColumnExprLiteral(ctx: ColumnExprLiteralContext): StringLiteral =
173     StringLiteral(Utils.stripSingleQuote(source(ctx.literal)))
174 
175   override def visitColumnExprFunction(ctx: ColumnExprFunctionContext): FuncExpr = {
176     require(
177       ctx.columnExprList == null,
178       s"Unsupported ColumnExprFunction with columnExprList: [${ctx.getClass.getSimpleName}] ${ctx.getText}"
179     )
180     require(
181       ctx.DISTINCT == null,
182       s"Unsupported ColumnExprFunction with DISTINCT: [${ctx.getClass.getSimpleName}] ${ctx.getText}"
183     )
184 
185     val funcName = ctx.identifier.getText
186     val funArgs = Option(ctx.columnArgList).map(_.columnArgExpr.asScala.toList.map { columnArgExprCtx =>
187       require(
188         columnArgExprCtx.columnLambdaExpr == null,
189         s"Unsupported ColumnLambdaExpr: ${source(columnArgExprCtx)}"
190       )
191       // recursive visit, but not sure if spark support the nested transform
192       visitColumnExpr(columnArgExprCtx.columnExpr)
193     }).getOrElse(Nil)
194     FuncExpr(funcName, funArgs)
195   }
196 
197   override def visitColumnExprTuple(ctx: ColumnExprTupleContext): TupleExpr =
198     TupleExpr(visitColumnExprList(ctx.columnExprList))
199 
200   override def visitColumnExprPrecedence1(ctx: ColumnExprPrecedence1Context): FuncExpr = {
201     val funcName =
202       if (ctx.PERCENT != null) "remainder"
203       else if (ctx.SLASH != null) "divide"
204       else if (ctx.ASTERISK != null) "multiply"
205       else throw new IllegalArgumentException(s"Invalid [ColumnExprPrecedence1] ${ctx.getText}")
206     val funArgs = List(visitColumnExpr(ctx.columnExpr(0)), visitColumnExpr(ctx.columnExpr(1)))
207     FuncExpr(funcName, funArgs)
208   }
209 
210   override def visitColumnExprPrecedence2(ctx: ColumnExprPrecedence2Context): FuncExpr = {
211     val funcName =
212       if (ctx.PLUS != null) "add"
213       else if (ctx.DASH != null) "subtract"
214       else if (ctx.CONCAT != null) "concat"
215       else throw new IllegalArgumentException(s"Invalid [ColumnExprPrecedence2] ${ctx.getText}")
216     val funArgs = List(visitColumnExpr(ctx.columnExpr(0)), visitColumnExpr(ctx.columnExpr(1)))
217     FuncExpr(funcName, funArgs)
218   }

224   override def visitOrderByClause(ctx: OrderByClauseContext): List[OrderExpr] =

227   override def visitOrderExpr(ctx: OrderExprContext): OrderExpr = {
228     val desc = Seq(ctx.DESC, ctx.DESCENDING).exists(_ != null)
229     val nullLast = Option(ctx.LAST).nonEmpty
230     OrderExpr(visitColumnExpr(ctx.columnExpr()), !desc, !nullLast)
231   }
232 
233   // //////////////////////////////////////////////
234   // /////////////// visit others /////////////////
235   // //////////////////////////////////////////////
236 
237   override def visitColumnExprList(ctx: ColumnExprListContext): List[Expr] =
238     ctx.columnsExpr.asScala.toList.map(visitColumnsExpr)
239 
240   def visitColumnsExpr(ctx: ColumnsExprContext): Expr = ctx match {
241     case field: ColumnsExprColumnContext => visitColumnExpr(field.columnExpr)
242     case other: ColumnsExprContext => throw new IllegalArgumentException(
243         s"Unsupported ColumnsExprContext: ${source(other)}"
244       )
245   }
246 
247   override def visitSettingsClause(ctx: SettingsClauseContext): Map[String, String] =
248     ctx.settingExprList.settingExpr.asScala.map(se => se.identifier.getText -> source(se.literal)).toMap
