56   protected lazy val shardExpr: Option[Expression] = writeJob.sparkShardExpr match {
57     case None => None
58     case Some(v2Expr) =>
59       val catalystExpr = ExprUtils.toCatalyst(v2Expr, writeJob.dataSetSchema.fields)
60       catalystExpr match {
61         case BoundReference(_, dataType, _)
62             if dataType.isInstanceOf[ByteType] // list all integral types here because we can not access `IntegralType`
63               || dataType.isInstanceOf[ShortType]
64               || dataType.isInstanceOf[IntegerType]
65               || dataType.isInstanceOf[LongType] =>
66           Some(catalystExpr)
67         case BoundReference(_, dataType, _) =>
68           throw CHClientException(s"Invalid data type of sharding field: $dataType")
69         case unsupported: Expression =>
70           log.warn(s"Unsupported expression of sharding field: $unsupported")
71           None
72       }
73   }

75   protected lazy val shardProjection: Option[expressions.Projection] = shardExpr
76     .filter(_ => writeJob.writeOptions.convertDistributedToLocal)
77     .map(expr => SafeProjection.create(Seq(expr)))

79   // put the node select strategy in executor side because we need to calculate shard and don't know the records
80   // util DataWriter#write(InternalRow) invoked.
81   protected lazy val client: Either[ClusterClient, NodeClient] =
82     writeJob.tableEngineSpec match {
83       case _: DistributedEngineSpec
84           if writeJob.writeOptions.useClusterNodesForDistributed || writeJob.writeOptions.convertDistributedToLocal =>
85         val clusterSpec = writeJob.cluster.get
86         log.info(s"Connect to cluster ${clusterSpec.name}, which has ${clusterSpec.shards.length} shards and " +
87           s"${clusterSpec.nodes.length} nodes.")
88         Left(ClusterClient(clusterSpec))
89       case _ =>
90         val nodeSpec = writeJob.node
91         log.info(s"Connect to single node: $nodeSpec")
92         Right(NodeClient(nodeSpec))
93     }
94 
95   def nodeClient(shardNum: Option[Int]): NodeClient = client match {
96     case Left(clusterClient) => clusterClient.node(shardNum)
97     case Right(nodeClient) => nodeClient
98   }

100   def calcShard(record: InternalRow): Option[Int] = (shardExpr, shardProjection) match {
101     case (Some(BoundReference(_, dataType, _)), Some(projection)) =>
102       val shardValue = dataType match {
103         case ByteType => Some(projection(record).getByte(0).toLong)
104         case ShortType => Some(projection(record).getShort(0).toLong)
105         case IntegerType => Some(projection(record).getInt(0).toLong)
106         case LongType => Some(projection(record).getLong(0))
107         case _ => None
108       }
109       shardValue.map(value => ShardUtils.calcShard(writeJob.cluster.get, value).num)
110     case _ => None
111   }

154   private def renewCompressedOutput(): Unit = {
155     val compressedOutput = (codec, protocol) match {
156       case (ClickHouseCompression.NONE, _) => observableSerializedOutput
157       case (ClickHouseCompression.LZ4, ClickHouseProtocol.HTTP) =>
158         // clickhouse http client forces compressed output stream
159         // new Lz4OutputStream(observableSerializedOutput, 4 * 1024 * 1024, null)
160         observableSerializedOutput
161       case unsupported =>
162         throw CHClientException(s"unsupported compression codec: $unsupported")
163     }
164     compressedForwardingOutput.updateDelegate(compressedOutput)
165   }

180   override def write(record: InternalRow): Unit = {
181     val shardNum = calcShard(record)
182     flush(force = shardNum != currentShardNum && currentBufferedRows > 0, currentShardNum)
183     currentShardNum = shardNum
184     val (_, serializedTime) = Utils.timeTakenMs(writeRow(record))
185     _lastSerializeTime.add(serializedTime)
186     _totalSerializeTime.add(serializedTime)
187     _currentBufferedRows.add(1)
188     flush(force = false, currentShardNum)
189   }

193   def serialize(): Array[Byte] = {
194     val (data, serializedTime) = Utils.timeTakenMs(doSerialize())
195     _lastSerializeTime.add(serializedTime)
196     _totalSerializeTime.add(serializedTime)
197     data
198   }

200   def doSerialize(): Array[Byte]
201 
202   def reset(): Unit = {
203     _currentBufferedRows.reset()
204     _currentRawBytesWritten.reset()
205     _lastSerializedBytesWritten.reset()
206     _lastSerializeTime.reset()
207     currentShardNum = None
208     serializedBuffer.reset()
209     renewCompressedOutput()
210   }
211 
212   def flush(force: Boolean, shardNum: Option[Int]): Unit =
213     if (force) {
214       doFlush(shardNum)
215     } else if (currentBufferedRows >= writeJob.writeOptions.batchSize) {
216       doFlush(shardNum)
217     }
218 
219   def doFlush(shardNum: Option[Int]): Unit = {
220     val client = nodeClient(shardNum)
221     val data = serialize()
222     var writeTime = 0L
223     Utils.retry[Unit, RetryableCHException](
224       writeJob.writeOptions.maxRetry,
225       writeJob.writeOptions.retryInterval
226     ) {
227       var startWriteTime = System.currentTimeMillis
228       client.syncInsertOutputJSONEachRow(database, table, format, codec, new ByteArrayInputStream(data)) match {
229         case Right(_) =>
230           writeTime = System.currentTimeMillis - startWriteTime
231           _totalWriteTime.add(writeTime)
232           _totalRecordsWritten.add(currentBufferedRows)
233         case Left(retryable) if writeJob.writeOptions.retryableErrorCodes.contains(retryable.code) =>
234           startWriteTime = System.currentTimeMillis
235           throw RetryableCHException(retryable.code, retryable.reason, Some(client.nodeSpec))
236         case Left(rethrow) => throw rethrow
237       }
238     } match {
239       case Success(_) =>
240         log.info(
241           s"""Job[${writeJob.queryId}]: batch write completed
242              |cluster: ${writeJob.cluster.map(_.name).getOrElse("none")}, shard: ${shardNum.getOrElse("none")}
243              |node: ${client.nodeSpec}
244              |        row count: $currentBufferedRows
245              |         raw size: ${Utils.bytesToString(currentBufferedRawBytes)}
246              |           format: $format
247              |compression codec: $codec
248              |  serialized size: ${Utils.bytesToString(lastSerializedBytesWritten)}
249              |   serialize time: ${lastSerializeTime}ms
250              |       write time: ${writeTime}ms
251              |""".stripMargin
252         )
253         reset()
254       case Failure(rethrow) => throw rethrow
255     }
256   }
257 
258   override def commit(): WriterCommitMessage = {
259     flush(currentBufferedRows > 0, currentShardNum)
260     CommitMessage(s"Job[${writeJob.queryId}]: commit")
261   }
262 
263   override def abort(): Unit = {}
264 
265   override def close(): Unit = {
266     IOUtils.closeQuietly(output)
267     client match {
268       case Left(clusterClient) => clusterClient.close()
269       case Right(nodeClient) => nodeClient.close()
270     }
271   }
