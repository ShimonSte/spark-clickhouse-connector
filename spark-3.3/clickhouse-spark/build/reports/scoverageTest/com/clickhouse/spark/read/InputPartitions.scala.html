27 case class ClickHouseInputPartition(
28   table: TableSpec,
29   partition: PartitionSpec,
30   filterByPartitionId: Boolean,
31   candidateNodes: Nodes, // try to use them only when preferredNode unavailable
32   preferredNode: Option[NodeSpec] = None // TODO assigned by ScanBuilder in Spark Driver side
33 ) extends InputPartition {
34 
35   override def preferredLocations(): Array[String] = preferredNode match {
36     case Some(preferred) => Array(preferred.host)
37     case None => candidateNodes.nodes.map(_.host)
38   }
39 
40   def partFilterExpr: String = partition match {
41     case NoPartitionSpec => "1=1"
42     case PartitionSpec(_, partitionId, _, _) if filterByPartitionId =>
43       s"_partition_id = '$partitionId'"
44     case PartitionSpec(partitionValue, _, _, _) =>
45       s"${table.partition_key} = ${compilePartitionFilterValue(partitionValue)}"
46   }
47 
48   // TODO improve and test
49   def compilePartitionFilterValue(partitionValue: String): String =
50     (partitionValue.contains("-"), partitionValue.contains("(")) match {
51       // quote when partition by a single Date Type column to avoid illegal types of arguments (Date, Int64)
52       case (true, false) => s"'$partitionValue'"
53       // Date type column is quoted if there are multi partition columns
54       case _ => s"$partitionValue"
55     }
