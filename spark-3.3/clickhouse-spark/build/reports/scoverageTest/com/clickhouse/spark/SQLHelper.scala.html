33   def compileValue(value: Any)(implicit tz: ZoneId): Any = value match {
34     case string: String => s"'${escapeSql(string)}'"
35     case utf8: UTF8String => s"'${escapeSql(utf8.toString)}'"
36     case instant: Instant => s"'${dateTimeFmt.withZone(tz).format(instant)}'"
37     case timestamp: Timestamp => s"'${legacyDateTimeFmt.format(timestamp)}'"
38     case localDateTime: LocalDateTime => s"'${dateTimeFmt.format(localDateTime)}'"
39     case legacyDate: Date => s"'${legacyDateFmt.format(legacyDate)}'"
40     case localDate: LocalDate => s"'${dateFmt.format(localDate)}'"
41     case array: Array[Any] => array.map(compileValue).mkString(",")
42     case _ => value
43   }

45   def compileFilter(f: Filter)(implicit tz: ZoneId): Option[String] = Option(f match {
46     case AlwaysTrue => "1=1"
47     case AlwaysFalse => "1=0"
48     case EqualTo(attr, value) => s"${quoted(attr)} = ${compileValue(value)}"
49     case EqualNullSafe(attr, nullableValue) =>
50       val (col, value) = (quoted(attr), compileValue(nullableValue))
51       s"(NOT ($col != $value OR $col IS NULL OR $value IS NULL) OR ($col IS NULL AND $value IS NULL))"
52     case LessThan(attr, value) => s"${quoted(attr)} < ${compileValue(value)}"
53     case GreaterThan(attr, value) => s"${quoted(attr)} > ${compileValue(value)}"
54     case LessThanOrEqual(attr, value) => s"${quoted(attr)} <= ${compileValue(value)}"
55     case GreaterThanOrEqual(attr, value) => s"${quoted(attr)} >= ${compileValue(value)}"
56     case IsNull(attr) => s"${quoted(attr)} IS NULL"
57     case IsNotNull(attr) => s"${quoted(attr)} IS NOT NULL"
58     case StringStartsWith(attr, value) => s"${quoted(attr)} LIKE '$value%'"
59     case StringEndsWith(attr, value) => s"${quoted(attr)} LIKE '%$value'"
60     case StringContains(attr, value) => s"${quoted(attr)} LIKE '%$value%'"
61     case In(attr, value) if value.isEmpty => s"CASE WHEN ${quoted(attr)} IS NULL THEN NULL ELSE FALSE END"
62     case In(attr, value) => s"${quoted(attr)} IN (${compileValue(value)})"
63     case Not(f) => compileFilter(f).map(p => s"(NOT ($p))").orNull
64     case Or(f1, f2) =>
65       val or = Seq(f1, f2).flatMap(_f => compileFilter(_f)(tz))
66       if (or.size == 2) or.map(p => s"($p)").mkString(" OR ") else null
67     case And(f1, f2) =>
68       val and = Seq(f1, f2).flatMap(_f => compileFilter(_f)(tz))
69       if (and.size == 2) and.map(p => s"($p)").mkString(" AND ") else null
70     case _ => null
71   })

73   def compileAggregate(aggFunction: AggregateFunc): Option[String] =
74     aggFunction match {
75       case min: Min if min.column.isInstanceOf[NamedReference] =>
76         val col = min.column.asInstanceOf[NamedReference]
77         if (col.fieldNames().length != 1) return None
78         Some(s"MIN(${quoted(col.fieldNames.head)})")
79       case max: Max if max.column.isInstanceOf[NamedReference] =>
80         val col = max.column.asInstanceOf[NamedReference]
81         if (col.fieldNames.length != 1) return None
82         Some(s"MAX(${quoted(col.fieldNames.head)})")
83       case count: Count if count.column.isInstanceOf[NamedReference] =>
84         val col = count.column.asInstanceOf[NamedReference]
85         if (col.fieldNames.length != 1) return None
86         val distinct = if (count.isDistinct) "DISTINCT " else ""
87         val column = quoted(col.fieldNames.head)
88         Some(s"COUNT($distinct$column)")
89       case sum: Sum if sum.column.isInstanceOf[NamedReference] =>
90         val col = sum.column.asInstanceOf[NamedReference]
91         if (col.fieldNames.length != 1) return None
92         val distinct = if (sum.isDistinct) "DISTINCT " else ""
93         val column = quoted(col.fieldNames.head)
94         Some(s"SUM($distinct$column)")
95       case _: CountStar =>
96         Some("COUNT(*)")
97       case _ => None
98     }
