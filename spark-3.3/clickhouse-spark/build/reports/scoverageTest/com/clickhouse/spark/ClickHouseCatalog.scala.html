116   override def loadTable(ident: Identifier): ClickHouseTable = {
117     val (database, table) = unwrap(ident) match {
118       case None => throw new NoSuchTableException(ident)
119       case Some((db, tbl)) =>
120         nodeClient.syncQueryOutputJSONEachRow(s"SELECT * FROM `$db`.`$tbl` WHERE 1=0") match {
121           case Left(exception) if exception.code == UNKNOWN_TABLE.code =>
122             throw new NoSuchTableException(ident)
123           // not sure if this check is necessary
124           case Left(exception) if exception.code == UNKNOWN_DATABASE.code =>
125             throw NoSuchTableException(s"Database $db does not exist")
126           case Left(rethrow) =>
127             throw rethrow
128           case Right(_) => (db, tbl)
129         }
130     }
131     implicit val _tz: ZoneId = tz.merge
132     val tableSpec = queryTableSpec(database, table)
133     val tableEngineSpec = TableEngineUtils.resolveTableEngine(tableSpec)
134     val tableClusterSpec = tableEngineSpec match {
135       case distributeSpec: DistributedEngineSpec =>
136         Some(TableEngineUtils.resolveTableCluster(distributeSpec, clusterSpecs))
137       case _ => None
138     }
139     ClickHouseTable(
140       nodeSpec,
141       tableClusterSpec,
142       _tz,
143       tableSpec,
144       tableEngineSpec
145     )

150    * {{{
151    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
152    * (
153    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1] [TTL expr1],
154    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2] [TTL expr2],
155    *     ...
156    *     INDEX index_name1 expr1 TYPE type1(...) GRANULARITY value1,
157    *     INDEX index_name2 expr2 TYPE type2(...) GRANULARITY value2
158    * ) ENGINE = MergeTree()
159    * ORDER BY expr
160    * [PARTITION BY expr]
161    * [PRIMARY KEY expr]
162    * [SAMPLE BY expr]
163    * [TTL expr
164    *     [DELETE|TO DISK 'xxx'|TO VOLUME 'xxx' [, ...] ]
165    *     [WHERE conditions]
166    *     [GROUP BY key_expr [SET v1 = agg_func(v1) [, v2 = agg_func(v2) ...]] ]]
167    * [SETTINGS name=value, ...]
168    * }}}
169    * <p>
170    *
171    * <h2>ReplacingMergeTree Engine</h2>
172    * {{{
173    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
174    * (
175    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1],
176    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2],
177    *     ...
178    * ) ENGINE = ReplacingMergeTree([ver])
179    * [PARTITION BY expr]
180    * [ORDER BY expr]
181    * [PRIMARY KEY expr]
182    * [SAMPLE BY expr]
183    * [SETTINGS name=value, ...]
184    * }}}
185    *
186    * `ver` — column with version. Type `UInt*`, `Date` or `DateTime`.

190   override def createTable(
191     ident: Identifier,
192     schema: StructType,
193     partitions: Array[Transform],
194     properties: util.Map[String, String]
195   ): ClickHouseTable = {
196     val (db, tbl) = unwrap(ident) match {
197       case Some((d, t)) => (d, t)
198       case None => throw CHClientException(s"Invalid table identifier: $ident")
199     }
200     val props = properties.asScala
201 
202     val engineExpr = props.getOrElse("engine", "MergeTree()")
203 
204     val isCreatingDistributed = engineExpr equalsIgnoreCase "Distributed"
205     val keyPrefix = if (isCreatingDistributed) "local." else ""
206 
207     val partitionsClause = partitions match {
208       case transforms if transforms.nonEmpty =>
209         transforms.map(ExprUtils.toClickHouse(_).sql).mkString("PARTITION BY (", ", ", ")")
210       case _ => ""
211     }
212 
213     val orderClause = props.get(s"${keyPrefix}order_by").map(o => s"ORDER BY ($o)").getOrElse("")
214     val primaryKeyClause = props.get(s"${keyPrefix}primary_key").map(p => s"PRIMARY KEY ($p)").getOrElse("")
215     val sampleClause = props.get(s"${keyPrefix}sample_by").map(p => s"SAMPLE BY ($p)").getOrElse("")
216 
217     val fieldsClause = SchemaUtils
218       .toClickHouseSchema(schema)
219       .map { case (fieldName, ckType, comment) => s"${quoted(fieldName)} $ckType$comment" }
220       .mkString(",\n ")
221 
222     val clusterOpt = props.get("cluster")
223 
224     def tblSettingsClause(prefix: String): String = props.filterKeys(_.startsWith(prefix)) match {
225       case settings if settings.nonEmpty =>
226         settings.map { case (k, v) =>
227           s"${k.substring(prefix.length)}=$v"
228         }.mkString("SETTINGS ", ", ", "")
229       case _ => ""
230     }
231 
232     def createTable(
233       clusterOpt: Option[String],
234       engineExpr: String,
235       database: String,
236       table: String,
237       settingsClause: String
238     ): Unit = {
239       val clusterClause = clusterOpt.map(c => s"ON CLUSTER $c").getOrElse("")
240       nodeClient.syncQueryAndCheckOutputJSONEachRow(
241         s"""CREATE TABLE `$database`.`$table` $clusterClause (
242            |$fieldsClause
243            |) ENGINE = $engineExpr
244            |$partitionsClause
245            |$orderClause
246            |$primaryKeyClause
247            |$sampleClause
248            |$settingsClause
249            |""".stripMargin
250           .replaceAll("""\n\s+\n""", "\n") // remove empty lines
251       )
252     }
253 
254     def createDistributedTable(
255       cluster: String,
256       shardExpr: String,
257       localDatabase: String,
258       localTable: String,
259       distributedDatabase: String,
260       distributedTable: String,
261       settingsClause: String
262     ): Unit = nodeClient.syncQueryAndCheckOutputJSONEachRow(
263       s"""CREATE TABLE `$distributedDatabase`.`$distributedTable` ON CLUSTER $cluster
264          |AS `$localDatabase`.`$localTable`
265          |ENGINE = Distributed($cluster, '$localDatabase', '$localTable', ($shardExpr))
266          |$settingsClause
267          |""".stripMargin
268     )
269 
270     if (isCreatingDistributed) {
271       val cluster = clusterOpt.getOrElse("default")
272       val shardExpr = props.getOrElse("shard_by", "rand()")
273       val settingsClause = tblSettingsClause("settings.")
274       val localEngineExpr = props.getOrElse(s"${keyPrefix}engine", s"MergeTree()")
275       val localDatabase = props.getOrElse(s"${keyPrefix}database", db)
276       val localTable = props.getOrElse(s"${keyPrefix}table", s"${tbl}_local")
277       val localSettingsClause = tblSettingsClause(s"${keyPrefix}settings.")
278       createTable(Some(cluster), localEngineExpr, localDatabase, localTable, localSettingsClause)
279       createDistributedTable(cluster, shardExpr, localDatabase, localTable, db, tbl, settingsClause)
280     } else {
281       val settingsClause = tblSettingsClause(s"${keyPrefix}settings.")
282       createTable(clusterOpt, engineExpr, db, tbl, settingsClause)
283     }
284 
285     loadTable(ident)
286   }

292   override def dropTable(ident: Identifier): Boolean = {
293     val tableOpt =
294       try Some(loadTable(ident))
295       catch {
296         case _: NoSuchTableException => None
297       }
298     tableOpt match {
299       case None => false
300       case Some(ClickHouseTable(_, cluster, _, tableSpec, _)) =>
301         val (db, tbl) = (tableSpec.database, tableSpec.name)
302         val isAtomic = loadNamespaceMetadata(Array(db)).get("engine").equalsIgnoreCase("atomic")
303         val syncClause = if (isAtomic) "SYNC" else ""
304         // limitation: only support Distribute table, can not handle cases such as drop local table on cluster nodes
305         val clusterClause = cluster.map(c => s"ON CLUSTER ${c.name}").getOrElse("")
306         nodeClient.syncQueryOutputJSONEachRow(s"DROP TABLE `$db`.`$tbl` $clusterClause $syncClause").isRight
307     }
308   }
