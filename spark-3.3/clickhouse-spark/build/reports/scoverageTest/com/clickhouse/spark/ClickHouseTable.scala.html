184   override def dropPartition(ident: InternalRow): Boolean = {
185     val partitionExpr = (0 until ident.numFields).map { i =>
186       partitionSchema.fields(i).dataType match {
187         case IntegerType => compileValue(ident.getInt(i))
188         case LongType => compileValue(ident.getLong(i))
189         case StringType => compileValue(ident.getUTF8String(i))
190         case DateType => compileValue(LocalDate.ofEpochDay(ident.getInt(i)))
191         case illegal => throw new IllegalArgumentException(s"Illegal partition data type: $illegal")
192       }
193     }.mkString("(", ",", ")")

196       engineSpec match {
197         case DistributedEngineSpec(_, cluster, local_db, local_table, _, _) =>
198           dropPartition(local_db, local_table, partitionExpr, Some(cluster))
199         case _ =>
200           dropPartition(database, table, partitionExpr)
201       }
202     }
203   }

215   override def listPartitionIdentifiers(names: Array[String], ident: InternalRow): Array[InternalRow] = {
216     assert(
217       names.length == ident.numFields,
218       s"Number of partition names (${names.length}) must be equal to " +
219         s"the number of partition values (${ident.numFields})."
220     )
221     assert(
222       names.forall(fieldName => partitionSchema.fieldNames.contains(fieldName)),
223       s"Some partition names ${names.mkString("[", ", ", "]")} don't belong to " +
224         s"the partition schema '${partitionSchema.sql}'."
225     )

227     def strToSparkValue(str: String, dataType: DataType): Any = dataType match {
228       case StringType => UTF8String.fromString(str.stripPrefix("'").stripSuffix("'"))
229       case IntegerType => JInt.parseInt(str)
230       case LongType => JLong.parseLong(str)
231       case DateType => LocalDate.parse(str.stripPrefix("'").stripSuffix("'"), dateFmt).toEpochDay.toInt
232       case unsupported => throw new UnsupportedOperationException(s"$unsupported")
233     }
234 
235     val partitionSpecs: Seq[PartitionSpec] = engineSpec match {
236       case DistributedEngineSpec(_, _, local_db, local_table, _, _) =>
237         cluster.get.shards.flatMap { shardSpec =>
238           Utils.tryWithResource(NodeClient(shardSpec.nodes.head)) { implicit nodeClient: NodeClient =>
239             queryPartitionSpec(local_db, local_table)
240           }
241         }
242       case _ =>
243         Utils.tryWithResource(NodeClient(node)) { implicit nodeClient =>
244           queryPartitionSpec(database, table)
245         }
246     }
247     partitionSpecs.map(_.partition_value)
248       .distinct
249       .filterNot(_.isEmpty) // represent partitioned table w/o records
250       .filterNot(_ == "tuple()") // represent the root partition of un-partitioned table
251       .map {
252         case tuple if tuple.startsWith("(") && tuple.endsWith(")") =>
253           tuple.stripPrefix("(").stripSuffix(")").split(",")
254         case partColStrValue =>
255           Array(partColStrValue)
256       }
257       .map { partColStrValues =>
258         new GenericInternalRow(
259           (partColStrValues zip partitionSchema.fields.map(_.dataType))
260             .map { case (partColStrValue, dataType) => strToSparkValue(partColStrValue, dataType) }
261         )
262       }
263       .filter { partRow =>
264         names.zipWithIndex.forall { case (name, queryIndex) =>
265           val partRowIndex = partitionSchema.fieldIndex(name)
266           val dataType = partitionSchema.fields(partRowIndex).dataType
267           partRow.get(partRowIndex, dataType) == ident.get(queryIndex, dataType)
268         }
269       }
270       .toArray
271   }
