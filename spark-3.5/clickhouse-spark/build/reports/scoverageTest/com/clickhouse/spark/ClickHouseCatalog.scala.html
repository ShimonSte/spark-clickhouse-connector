159    * <h2>MergeTree Engine</h2>
160    * {{{
161    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
162    * (
163    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1] [TTL expr1],
164    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2] [TTL expr2],
165    *     ...
166    *     INDEX index_name1 expr1 TYPE type1(...) GRANULARITY value1,
167    *     INDEX index_name2 expr2 TYPE type2(...) GRANULARITY value2
168    * ) ENGINE = MergeTree()
169    * ORDER BY expr
170    * [PARTITION BY expr]
171    * [PRIMARY KEY expr]
172    * [SAMPLE BY expr]
173    * [TTL expr
174    *     [DELETE|TO DISK 'xxx'|TO VOLUME 'xxx' [, ...] ]
175    *     [WHERE conditions]
176    *     [GROUP BY key_expr [SET v1 = agg_func(v1) [, v2 = agg_func(v2) ...]] ]]
177    * [SETTINGS name=value, ...]
178    * }}}
179    * <p>
180    *
181    * <h2>ReplacingMergeTree Engine</h2>
182    * {{{
183    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
184    * (
185    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1],
186    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2],
187    *     ...
188    * ) ENGINE = ReplacingMergeTree([ver])
189    * [PARTITION BY expr]
190    * [ORDER BY expr]
191    * [PRIMARY KEY expr]
192    * [SAMPLE BY expr]
193    * [SETTINGS name=value, ...]
194    * }}}
195    *
196    * `ver` — column with version. Type `UInt*`, `Date` or `DateTime`.

200   override def createTable(
201     ident: Identifier,
202     schema: StructType,
203     partitions: Array[Transform],
204     properties: util.Map[String, String]
205   ): ClickHouseTable = {
206     val (db, tbl) = unwrap(ident) match {
207       case Some((d, t)) => (d, t)
208       case None => throw CHClientException(s"Invalid table identifier: $ident")
209     }
210     val props = properties.asScala
211 
212     val engineExpr = props.getOrElse("engine", "MergeTree()")
213 
214     val isCreatingDistributed = engineExpr equalsIgnoreCase "Distributed"
215     val keyPrefix = if (isCreatingDistributed) "local." else ""
216 
217     val partitionsClause = partitions match {
218       case transforms if transforms.nonEmpty =>
219         transforms.map(ExprUtils.toClickHouse(_, functionRegistry).sql).mkString("PARTITION BY (", ", ", ")")
220       case _ => ""
221     }
222 
223     val orderClause = props.get(s"${keyPrefix}order_by").map(o => s"ORDER BY ($o)").getOrElse("")
224     val primaryKeyClause = props.get(s"${keyPrefix}primary_key").map(p => s"PRIMARY KEY ($p)").getOrElse("")
225     val sampleClause = props.get(s"${keyPrefix}sample_by").map(p => s"SAMPLE BY ($p)").getOrElse("")
226 
227     val fieldsClause = SchemaUtils
228       .toClickHouseSchema(schema)
229       .map { case (fieldName, ckType, comment) => s"${quoted(fieldName)} $ckType$comment" }
230       .mkString(",\n ")
231 
232     val clusterOpt = props.get("cluster")
233 
234     def tblSettingsClause(prefix: String): String = props.filterKeys(_.startsWith(prefix)) match {
235       case settings if settings.nonEmpty =>
236         settings.map { case (k, v) =>
237           s"${k.substring(prefix.length)}=$v"
238         }.mkString("SETTINGS ", ", ", "")
239       case _ => ""
240     }
241 
242     def createTable(
243       clusterOpt: Option[String],
244       engineExpr: String,
245       database: String,
246       table: String,
247       settingsClause: String
248     ): Unit = {
249       val clusterClause = clusterOpt.map(c => s"ON CLUSTER $c").getOrElse("")
250       nodeClient.syncQueryAndCheckOutputJSONEachRow(
251         s"""CREATE TABLE `$database`.`$table` $clusterClause (
252            |$fieldsClause
253            |) ENGINE = $engineExpr
254            |$partitionsClause
255            |$orderClause
256            |$primaryKeyClause
257            |$sampleClause
258            |$settingsClause
259            |""".stripMargin
260           .replaceAll("""\n\s+\n""", "\n") // remove empty lines
261       )
262     }
263 
264     def createDistributedTable(
265       cluster: String,
266       shardExpr: String,
267       localDatabase: String,
268       localTable: String,
269       distributedDatabase: String,
270       distributedTable: String,
271       settingsClause: String
272     ): Unit = nodeClient.syncQueryAndCheckOutputJSONEachRow(
273       s"""CREATE TABLE `$distributedDatabase`.`$distributedTable` ON CLUSTER $cluster
274          |AS `$localDatabase`.`$localTable`
275          |ENGINE = Distributed($cluster, '$localDatabase', '$localTable', ($shardExpr))
276          |$settingsClause
277          |""".stripMargin
278     )
279 
280     if (isCreatingDistributed) {
281       val cluster = clusterOpt.getOrElse("default")
282       val shardExpr = props.getOrElse("shard_by", "rand()")
283       val settingsClause = tblSettingsClause("settings.")
284       val localEngineExpr = props.getOrElse(s"${keyPrefix}engine", s"MergeTree()")
285       val localDatabase = props.getOrElse(s"${keyPrefix}database", db)
286       val localTable = props.getOrElse(s"${keyPrefix}table", s"${tbl}_local")
287       val localSettingsClause = tblSettingsClause(s"${keyPrefix}settings.")
288       createTable(Some(cluster), localEngineExpr, localDatabase, localTable, localSettingsClause)
289       createDistributedTable(cluster, shardExpr, localDatabase, localTable, db, tbl, settingsClause)
290     } else {
291       val settingsClause = tblSettingsClause(s"${keyPrefix}settings.")
292       createTable(clusterOpt, engineExpr, db, tbl, settingsClause)
293     }
294 
295     loadTable(ident)
296   }

302   override def dropTable(ident: Identifier): Boolean = {
303     val tableOpt =
304       try Some(loadTable(ident))
305       catch {
306         case _: NoSuchTableException => None
307       }
308     tableOpt match {
309       case None => false
310       case Some(ClickHouseTable(_, cluster, _, tableSpec, _, _)) =>
311         val (db, tbl) = (tableSpec.database, tableSpec.name)
312         val isAtomic = loadNamespaceMetadata(Array(db)).get("engine").equalsIgnoreCase("atomic")
313         val syncClause = if (isAtomic) "SYNC" else ""
314         // limitation: only support Distribute table, can not handle cases such as drop local table on cluster nodes
315         val clusterClause = cluster.map(c => s"ON CLUSTER ${c.name}").getOrElse("")
316         nodeClient.syncQueryOutputJSONEachRow(s"DROP TABLE `$db`.`$tbl` $clusterClause $syncClause").isRight
317     }
