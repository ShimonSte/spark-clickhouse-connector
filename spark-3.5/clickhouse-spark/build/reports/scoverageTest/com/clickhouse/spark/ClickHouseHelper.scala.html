58   def buildNodeSpec(options: CaseInsensitiveStringMap): NodeSpec = {
59     val clientOpts = options.asScala
60       .filterKeys(_.startsWith(CATALOG_PROP_OPTION_PREFIX))
61       .map { case (k, v) => k.substring(CATALOG_PROP_OPTION_PREFIX.length) -> v }
62       .toMap
63       .filterKeys { key =>
64         val ignore = CATALOG_PROP_IGNORE_OPTIONS.contains(key)
65         if (ignore) {
66           log.warn(s"Ignore configuration $key.")
67         }
68         !ignore
69       }
70       .toMap
71     NodeSpec(
72       _host = options.getOrDefault(CATALOG_PROP_HOST, "localhost"),
73       _tcp_port = Some(options.getInt(CATALOG_PROP_TCP_PORT, 9000)),
74       _http_port = Some(options.getInt(CATALOG_PROP_HTTP_PORT, 8123)),
75       protocol = ClickHouseProtocol.fromUriScheme(options.getOrDefault(CATALOG_PROP_PROTOCOL, "http")),
76       username = options.getOrDefault(CATALOG_PROP_USER, "default"),
77       password = options.getOrDefault(CATALOG_PROP_PASSWORD, ""),
78       database = options.getOrDefault(CATALOG_PROP_DATABASE, "default"),
79       options = new JHashMap(clientOpts.asJava)
80     )
81   }

83   def queryClusterSpecs(nodeSpec: NodeSpec)(implicit nodeClient: NodeClient): Seq[ClusterSpec] = {
84     val clustersOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
85       """ SELECT
86         |   `cluster`,                 -- String
87         |   `shard_num`,               -- UInt32
88         |   `shard_weight`,            -- UInt32
89         |   `replica_num`,             -- UInt32
90         |   `host_name`,               -- String
91         |   `host_address`,            -- String
92         |   `port`,                    -- UInt16
93         |   `is_local`,                -- UInt8
94         |   `user`,                    -- String
95         |   `default_database`,        -- String
96         |   `errors_count`,            -- UInt32
97         |   `estimated_recovery_time`  -- UInt32
98         | FROM `system`.`clusters`
99         |""".stripMargin
100     )
101     clustersOutput.records
102       .groupBy(_.get("cluster").asText)
103       .map { case (cluster, rows) =>
104         val shards = rows
105           .groupBy(_.get("shard_num").asInt)
106           .map { case (shardNum, rows) =>
107             val shardWeight = rows.head.get("shard_weight").asInt
108             val nodes = rows.map { row =>
109               val replicaNum = row.get("replica_num").asInt
110               // should other properties be provided by `SparkConf`?
111               val clickhouseNode = nodeSpec.copy(
112                 // host_address is not works for testcontainers
113                 _host = row.get("host_name").asText,
114                 _tcp_port = Some(row.get("port").asInt),
115                 _http_port = if (Utils.isTesting) Some(8123) else nodeSpec.http_port
116               )
117               ReplicaSpec(replicaNum, clickhouseNode)
118             }.toArray
119             ShardSpec(shardNum, shardWeight, nodes)
120           }.toArray
121         ClusterSpec(cluster, shards)
122       }.toSeq
123   }

203       ),
204       dependencies_database = tableRow.get("dependencies_database").elements().asScala.map(_.asText).toList,
205       dependencies_table = tableRow.get("dependencies_table").elements().asScala.map(_.asText).toList,
206       create_table_query = tableRow.get("create_table_query").asText,
207       engine_full = tableRow.get("engine_full").asText,
208       partition_key = tableRow.get("partition_key").asText,
209       sorting_key = tableRow.get("sorting_key").asText,
210       primary_key = tableRow.get("primary_key").asText,
211       sampling_key = tableRow.get("sampling_key").asText,
212       storage_policy = tableRow.get("storage_policy").asText,

214         case _: NullNode | null => None
215         case node: JsonNode => Some(node.asLong)
216       },
217       total_bytes = tableRow.get("total_bytes") match {
218         case _: NullNode | null => None

220       },

224       },

226         case _: NullNode | null => None
227         case node: JsonNode => Some(node.asLong)
228       }
229     )
230   }

233     database: String,
234     table: String,
235     actionIfNoSuchTable: (String, String) => Unit = DEFAULT_ACTION_IF_NO_SUCH_TABLE
236   )(implicit nodeClient: NodeClient): StructType = {

238       s"""SELECT
239          |  `database`,                -- String
240          |  `table`,                   -- String
241          |  `name`,                    -- String
242          |  `type`,                    -- String
243          |  `position`,                -- UInt64
244          |  `default_kind`,            -- String

270   def queryPartitionSpec(
271     database: String,
272     table: String
273   )(implicit nodeClient: NodeClient): Seq[PartitionSpec] = {
274     val partOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
275       s"""SELECT
276          |  partition,                           -- String
277          |  partition_id,                        -- String
278          |  sum(rows)          AS row_count,     -- UInt64
279          |  sum(bytes_on_disk) AS size_in_bytes  -- UInt64
280          |FROM `system`.`parts`
281          |WHERE `database`='$database' AND `table`='$table' AND `active`=1
282          |GROUP BY `partition`, `partition_id`
283          |ORDER BY `partition` ASC, partition_id ASC
284          |""".stripMargin
285     )
286     if (partOutput.isEmpty || partOutput.rows == 1 && partOutput.records.head.get("partition").asText == "tuple()") {
287       return Array(NoPartitionSpec)
288     }
289     partOutput.records.map { row =>
290       PartitionSpec(
291         partition_value = row.get("partition").asText,
292         partition_id = row.get("partition_id").asText,
293         row_count = row.get("row_count").asLong,
294         size_in_bytes = row.get("size_in_bytes").asLong
295       )
296     }
297   }

308   def dropPartition(
309     database: String,
310     table: String,
311     partitionExpr: String,
312     cluster: Option[String] = None
313   )(implicit
314     nodeClient: NodeClient
315   ): Boolean =
316     nodeClient.syncQueryOutputJSONEachRow(
317       s"ALTER TABLE `$database`.`$table` ${cluster.map(c => s"ON CLUSTER $c").getOrElse("")} DROP PARTITION $partitionExpr"
318     ) match {
319       case Right(_) => true
320       case Left(ex: CHException) =>
321         log.error(s"[${ex.code}]: ${ex.getMessage}")
322         false
