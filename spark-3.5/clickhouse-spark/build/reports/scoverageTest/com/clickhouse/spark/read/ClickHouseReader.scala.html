29 abstract class ClickHouseReader[Record](

37   val readDistributedUseClusterNodes: Boolean = conf.getConf(READ_DISTRIBUTED_USE_CLUSTER_NODES)
38   val readDistributedConvertLocal: Boolean = conf.getConf(READ_DISTRIBUTED_CONVERT_LOCAL)
39   private val readSettings: Option[String] = conf.getConf(READ_SETTINGS)
40 
41   val database: String = part.table.database
42   val table: String = part.table.name
43   val codec: ClickHouseCompression = scanJob.readOptions.compressionCodec
44   val readSchema: StructType = scanJob.readSchema
45 
46   private lazy val nodesClient = NodesClient(part.candidateNodes)
47 
48   def nodeClient: NodeClient = nodesClient.node
49 
50   lazy val scanQuery: String = {
51     val selectItems =
52       if (readSchema.isEmpty) {
53         "1" // for case like COUNT(*) which prunes all columns
54       } else {
55         readSchema.map {
56           field => if (scanJob.groupByClause.isDefined) field.name else s"`${field.name}`"
57         }.mkString(", ")
58       }
59     s"""SELECT $selectItems
60        |FROM `$database`.`$table`
61        |WHERE (${part.partFilterExpr}) AND (${scanJob.filtersExpr})
62        |${scanJob.groupByClause.getOrElse("")}
63        |${scanJob.limit.map(n => s"LIMIT $n").getOrElse("")}
64        |${readSettings.map(settings => s"SETTINGS $settings").getOrElse("")}
65        |""".stripMargin
66   }
67 
68   def format: String
69 
70   lazy val resp: ClickHouseResponse = nodeClient.queryAndCheck(scanQuery, format, codec)
71 
72   def totalBlocksRead: Long = resp.getSummary.getStatistics.getBlocks
73 
74   def totalBytesRead: Long = resp.getSummary.getReadBytes
75 
76   override def currentMetricsValues: Array[CustomTaskMetric] = Array(
77     TaskMetric(BLOCKS_READ, totalBlocksRead),
78     TaskMetric(BYTES_READ, totalBytesRead)
79   )
80 
81   def streamOutput: Iterator[Record]
82 
83   private var currentRecord: Record = _
84 
85   override def next(): Boolean = {
86     val hasNext = streamOutput.hasNext
87     if (hasNext) currentRecord = streamOutput.next
88     hasNext
89   }
90 
91   override def get: InternalRow = decode(currentRecord)

93   def decode(record: Record): InternalRow
94 
95   override def close(): Unit = nodesClient.close()
96 }
