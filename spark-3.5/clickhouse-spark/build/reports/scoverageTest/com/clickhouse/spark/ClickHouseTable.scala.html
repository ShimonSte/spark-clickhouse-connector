198   override def dropPartition(ident: InternalRow): Boolean = {
199     val partitionExpr = (0 until ident.numFields).map { i =>
200       partitionSchema.fields(i).dataType match {
201         case IntegerType => compileValue(ident.getInt(i))
202         case LongType => compileValue(ident.getLong(i))
203         case StringType => compileValue(ident.getUTF8String(i))
204         case DateType => compileValue(LocalDate.ofEpochDay(ident.getInt(i)))
205         case illegal => throw new IllegalArgumentException(s"Illegal partition data type: $illegal")
206       }
207     }.mkString("(", ",", ")")

210       engineSpec match {
211         case DistributedEngineSpec(_, cluster, local_db, local_table, _, _) =>
212           dropPartition(local_db, local_table, partitionExpr, Some(cluster))
213         case _ =>
214           dropPartition(database, table, partitionExpr)
215       }
216     }
217   }

229   override def listPartitionIdentifiers(names: Array[String], ident: InternalRow): Array[InternalRow] = {
230     assert(
231       names.length == ident.numFields,
232       s"Number of partition names (${names.length}) must be equal to " +
233         s"the number of partition values (${ident.numFields})."
234     )
235     assert(
236       names.forall(fieldName => partitionSchema.fieldNames.contains(fieldName)),
237       s"Some partition names ${names.mkString("[", ", ", "]")} don't belong to " +
238         s"the partition schema '${partitionSchema.sql}'."
239     )

241     def strToSparkValue(str: String, dataType: DataType): Any = dataType match {
242       case StringType => UTF8String.fromString(str.stripPrefix("'").stripSuffix("'"))
243       case IntegerType => JInt.parseInt(str)
244       case LongType => JLong.parseLong(str)
245       case DateType => LocalDate.parse(str.stripPrefix("'").stripSuffix("'"), dateFmt).toEpochDay.toInt
246       case unsupported => throw new UnsupportedOperationException(s"$unsupported")
247     }
248 
249     val partitionSpecs: Seq[PartitionSpec] = engineSpec match {
250       case DistributedEngineSpec(_, _, local_db, local_table, _, _) =>
251         cluster.get.shards.flatMap { shardSpec =>
252           Utils.tryWithResource(NodeClient(shardSpec.nodes.head)) { implicit nodeClient: NodeClient =>
253             queryPartitionSpec(local_db, local_table)
254           }
255         }
256       case _ =>
257         Utils.tryWithResource(NodeClient(node)) { implicit nodeClient =>
258           queryPartitionSpec(database, table)
259         }
260     }
261     partitionSpecs.map(_.partition_value)
262       .distinct
263       .filterNot(_.isEmpty) // represent partitioned table w/o records
264       .filterNot(_ == "tuple()") // represent the root partition of un-partitioned table
265       .map {
266         case tuple if tuple.startsWith("(") && tuple.endsWith(")") =>
267           tuple.stripPrefix("(").stripSuffix(")").split(",")
268         case partColStrValue =>
269           Array(partColStrValue)
270       }
271       .map { partColStrValues =>
272         new GenericInternalRow(
273           (partColStrValues zip partitionSchema.fields.map(_.dataType))
274             .map { case (partColStrValue, dataType) => strToSparkValue(partColStrValue, dataType) }
275         )
276       }
277       .filter { partRow =>
278         names.zipWithIndex.forall { case (name, queryIndex) =>
279           val partRowIndex = partitionSchema.fieldIndex(name)
280           val dataType = partitionSchema.fields(partRowIndex).dataType
281           partRow.get(partRowIndex, dataType) == ident.get(queryIndex, dataType)
282         }
283       }
284       .toArray
285   }
