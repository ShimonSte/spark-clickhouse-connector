74   private def loadV2FunctionOpt(
75     name: String,
76     args: Seq[Expression],
77     functionRegistry: FunctionRegistry
78   ): Option[BoundFunction] = {
79     def loadFunction(ident: Identifier): UnboundFunction =
80       functionRegistry.load(ident.name).getOrElse(throw new NoSuchFunctionException(ident))
81     val inputType = StructType(args.zipWithIndex.map {
82       case (exp, pos) => StructField(s"_$pos", exp.dataType, exp.nullable)

84     try {
85       val unbound = loadFunction(Identifier.of(Array.empty, name))
86       Some(unbound.bind(inputType))

90       case _: UnsupportedOperationException if conf.getConf(IGNORE_UNSUPPORTED_TRANSFORM) =>
91         None
92       case e: UnsupportedOperationException =>

94     }

109   private class CustomResolveTimeZone(timeZoneId: Option[String]) extends Rule[LogicalPlan] {
110     override def apply(plan: LogicalPlan): LogicalPlan =
111       plan.resolveExpressionsWithPruning(_.containsAnyPattern(LIST_SUBQUERY, TIME_ZONE_AWARE_EXPRESSION)) {
112         case e: TimeZoneAwareExpression if e.timeZoneId.isEmpty =>
113           e.withTimeZone(timeZoneId.getOrElse(conf.sessionLocalTimeZone))
114         // Casts could be added in the subquery plan through the rule TypeCoercion while coercing
115         // the types between the value expression and list query expression of IN expression.
116         // We need to subject the subquery plan through ResolveTimeZone again to setup timezone
117         // information for time zone aware expressions.
118         case e: ListQuery => e.withNewPlan(apply(e.plan))
119       }
120   }

141   def toCatalyst(
142     v2Expr: V2Expression,
143     fields: Array[StructField],
144     functionRegistry: FunctionRegistry
145   ): Expression =
146     v2Expr match {
147       case IdentityTransform(ref) => toCatalyst(ref, fields, functionRegistry)
148       case ref: NamedReference if ref.fieldNames.length == 1 =>
149         val (field, ordinal) = fields
150           .zipWithIndex
151           .find { case (field, _) => field.name == ref.fieldNames.head }
152           .getOrElse(throw CHClientException(s"Invalid field reference: $ref"))
153         BoundReference(ordinal, field.dataType, field.nullable)
154       case t: Transform =>
155         val catalystArgs = t.arguments().map(toCatalyst(_, fields, functionRegistry))
156         loadV2FunctionOpt(t.name(), catalystArgs, functionRegistry)
157           .map(bound => TransformExpression(bound, catalystArgs)).getOrElse {
158             throw CHClientException(s"Unsupported expression: $v2Expr")
159           }
160       case literal: LiteralValue[Any] => expressions.Literal(literal.value)
161       case _ => throw CHClientException(
162           s"Unsupported expression: $v2Expr"
163         )
164     }

166   def toSparkTransformOpt(expr: Expr, functionRegistry: FunctionRegistry): Option[Transform] =
167     Try(toSparkExpression(expr, functionRegistry)) match {
168       // need this function because spark `Table`'s `partitioning` field should be `Transform`
169       case Success(t: Transform) => Some(t)
170       case Success(_) => None
171       case Failure(_) if conf.getConf(IGNORE_UNSUPPORTED_TRANSFORM) => None
172       case Failure(rethrow) => throw new AnalysisException(rethrow.getMessage, cause = Some(rethrow))
173     }

175   def toSparkExpression(expr: Expr, functionRegistry: FunctionRegistry): V2Expression =
176     expr match {
177       case FieldRef(col) => identity(col)
178       case StringLiteral(value) => literal(value) // TODO LiteralTransform
179       case FuncExpr("rand", Nil) => apply("rand")
180       case FuncExpr("toYYYYMMDD", List(FuncExpr("toDate", List(FieldRef(col))))) => identity(col)
181       case FuncExpr(funName, args) if functionRegistry.clickHouseToSparkFunc.contains(funName) =>
182         apply(functionRegistry.clickHouseToSparkFunc(funName), args.map(toSparkExpression(_, functionRegistry)): _*)
183       case unsupported => throw CHClientException(s"Unsupported ClickHouse expression: $unsupported")
184     }

186   def toClickHouse(
187     transform: Transform,
188     functionRegistry: FunctionRegistry
189   ): Expr = transform match {
190     case IdentityTransform(fieldRefs) => FieldRef(fieldRefs.describe)
191     case ApplyTransform(name, args) if functionRegistry.sparkToClickHouseFunc.contains(name) =>
192       FuncExpr(functionRegistry.sparkToClickHouseFunc(name), args.map(arg => SQLExpr(arg.describe)).toList)
193     case bucket: BucketTransform => throw CHClientException(s"Bucket transform not support yet: $bucket")
194     case other: Transform => throw CHClientException(s"Unsupported transform: $other")
195   }

197   def inferTransformSchema(
198     primarySchema: StructType,
199     secondarySchema: StructType,
200     transform: Transform,
201     functionRegistry: FunctionRegistry
202   ): StructField = transform match {
203     case IdentityTransform(FieldReference(Seq(col))) => primarySchema.find(_.name == col)
204         .orElse(secondarySchema.find(_.name == col))
205         .getOrElse(throw CHClientException(s"Invalid partition column: $col"))
206     case t @ ApplyTransform(transformName, _) if functionRegistry.load(transformName).isDefined =>
207       val resType = functionRegistry.load(transformName) match {
208         case Some(f: ScalarFunction[_]) => f.resultType
209         case other => throw CHClientException(s"Unsupported function: $other")
210       }
211       StructField(t.toString, resType)
212     case bucket: BucketTransform => throw CHClientException(s"Bucket transform not support yet: $bucket")
213     case other: Transform => throw CHClientException(s"Unsupported transform: $other")
214   }
