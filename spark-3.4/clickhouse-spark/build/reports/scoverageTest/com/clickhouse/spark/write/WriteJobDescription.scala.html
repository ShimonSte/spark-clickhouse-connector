25 case class WriteJobDescription(

27   tableSchema: StructType,

34   cluster: Option[ClusterSpec],
35   localTableSpec: Option[TableSpec],
36   localTableEngineSpec: Option[TableEngineSpec],
37   shardingKey: Option[Expr],
38   partitionKey: Option[List[Expr]],
39   sortingKey: Option[List[OrderExpr]],
40   writeOptions: WriteOptions,
41   functionRegistry: FunctionRegistry
42 ) {
43 
44   def targetDatabase(convert2Local: Boolean): String = tableEngineSpec match {
45     case dist: DistributedEngineSpec if convert2Local => dist.local_db
46     case _ => tableSpec.database
47   }
48 
49   def targetTable(convert2Local: Boolean): String = tableEngineSpec match {
50     case dist: DistributedEngineSpec if convert2Local => dist.local_table
51     case _ => tableSpec.name
52   }
53 
54   def shardingKeyIgnoreRand: Option[Expr] = shardingKey filter {
55     case FuncExpr("rand", Nil) => false
56     case _ => true
57   }
58 
59   def sparkShardExpr: Option[Expression] = shardingKeyIgnoreRand match {
60     case Some(expr) => ExprUtils.toSparkTransformOpt(expr, functionRegistry)
61     case _ => None
62   }
63 
64   def sparkSplits: Array[Transform] =
65     if (writeOptions.repartitionByPartition) {
66       ExprUtils.toSparkSplits(
67         shardingKeyIgnoreRand,
68         partitionKey,
69         functionRegistry
70       )
71     } else {
72       ExprUtils.toSparkSplits(
73         shardingKeyIgnoreRand,
74         None,
75         functionRegistry
76       )
77     }
78 
79   def sparkSortOrders: Array[SortOrder] = {
80     val _partitionKey = if (writeOptions.localSortByPartition) partitionKey else None
81     val _sortingKey = if (writeOptions.localSortByKey) sortingKey else None
82     ExprUtils.toSparkSortOrders(shardingKeyIgnoreRand, _partitionKey, _sortingKey, cluster, functionRegistry)
83   }
