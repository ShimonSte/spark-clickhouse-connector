57   protected lazy val shardExpr: Option[Expression] = writeJob.sparkShardExpr match {
58     case None => None
59     case Some(v2Expr) =>
60       val catalystExpr = ExprUtils.toCatalyst(v2Expr, writeJob.dataSetSchema.fields, writeJob.functionRegistry)
61       catalystExpr match {
62         case BoundReference(_, dataType, _)
63             if dataType.isInstanceOf[ByteType] // list all integral types here because we can not access `IntegralType`
64               || dataType.isInstanceOf[ShortType]
65               || dataType.isInstanceOf[IntegerType]
66               || dataType.isInstanceOf[LongType] =>
67           Some(catalystExpr)
68         case BoundReference(_, dataType, _) =>
69           throw CHClientException(s"Invalid data type of sharding field: $dataType")
70         case TransformExpression(function, _, _) =>
71           function.resultType match {
72             case ByteType | ShortType | IntegerType | LongType => Some(catalystExpr)
73             case _ => throw CHClientException(s"Invalid data type of sharding field: ${function.resultType}")
74           }
75         case unsupported: Expression =>
76           log.warn(s"Unsupported expression of sharding field: $unsupported")
77           None
78       }
79   }
80 
81   protected lazy val shardProjection: Option[expressions.Projection] = shardExpr
82     .filter(_ => writeJob.writeOptions.convertDistributedToLocal)
83     .flatMap {
84       case expr: BoundReference =>
85         Some(SafeProjection.create(Seq(expr)))
86       case expr @ TransformExpression(function, _, _) =>
87         // result type must be integer class
88         function.resultType match {
89           case ByteType => classOf[Byte]
90           case ShortType => classOf[Short]
91           case IntegerType => classOf[Int]
92           case LongType => classOf[Long]
93           case _ => throw CHClientException(s"Invalid return data type for function ${function.name()}," +
94               s"sharding field: ${function.resultType}")
95         }
96         Some(SafeProjection.create(Seq(ExprUtils.resolveTransformCatalyst(expr, Some(writeJob.tz.getId)))))
97     }

99   // put the node select strategy in executor side because we need to calculate shard and don't know the records
100   // util DataWriter#write(InternalRow) invoked.
101   protected lazy val client: Either[ClusterClient, NodeClient] =
102     writeJob.tableEngineSpec match {
103       case _: DistributedEngineSpec
104           if writeJob.writeOptions.useClusterNodesForDistributed || writeJob.writeOptions.convertDistributedToLocal =>
105         val clusterSpec = writeJob.cluster.get
106         log.info(s"Connect to cluster ${clusterSpec.name}, which has ${clusterSpec.shards.length} shards and " +
107           s"${clusterSpec.nodes.length} nodes.")
108         Left(ClusterClient(clusterSpec))
109       case _ =>
110         val nodeSpec = writeJob.node
111         log.info(s"Connect to single node: $nodeSpec")
112         Right(NodeClient(nodeSpec))
113     }

115   def nodeClient(shardNum: Option[Int]): NodeClient = client match {
116     case Left(clusterClient) => clusterClient.node(shardNum)
117     case Right(nodeClient) => nodeClient

120   def calcShard(record: InternalRow): Option[Int] = (shardExpr, shardProjection) match {
121     case (Some(BoundReference(_, dataType, _)), Some(projection)) =>
122       doCalcShard(record, dataType, projection)
123     case (Some(TransformExpression(function, _, _)), Some(projection)) =>
124       doCalcShard(record, function.resultType, projection)
125     case _ => None
126   }
127 
128   private def doCalcShard(record: InternalRow, dataType: DataType, projection: Projection): Option[Int] = {
129     val shardValue = dataType match {
130       case ByteType => Some(projection(record).getByte(0).toLong)
131       case ShortType => Some(projection(record).getShort(0).toLong)
132       case IntegerType => Some(projection(record).getInt(0).toLong)
133       case LongType => Some(projection(record).getLong(0))
134       case _ => None
135     }
136     shardValue.map(value => ShardUtils.calcShard(writeJob.cluster.get, value).num)
137   }

180   private def renewCompressedOutput(): Unit = {
181     val compressedOutput = (codec, protocol) match {
182       case (ClickHouseCompression.NONE, _) => observableSerializedOutput
183       case (ClickHouseCompression.LZ4, ClickHouseProtocol.HTTP) =>
184         // clickhouse http client forces compressed output stream
185         // new Lz4OutputStream(observableSerializedOutput, 4 * 1024 * 1024, null)
186         observableSerializedOutput

189     }
190     compressedForwardingOutput.updateDelegate(compressedOutput)
191   }

206   override def write(record: InternalRow): Unit = {
207     val shardNum = calcShard(record)
208     flush(force = shardNum != currentShardNum && currentBufferedRows > 0, currentShardNum)
209     currentShardNum = shardNum
210     val (_, serializedTime) = Utils.timeTakenMs(writeRow(record))
211     _lastSerializeTime.add(serializedTime)
212     _totalSerializeTime.add(serializedTime)
213     _currentBufferedRows.add(1)
214     flush(force = false, currentShardNum)
215   }

219   def serialize(): Array[Byte] = {
220     val (data, serializedTime) = Utils.timeTakenMs(doSerialize())
221     _lastSerializeTime.add(serializedTime)
222     _totalSerializeTime.add(serializedTime)
223     data
224   }

228   def reset(): Unit = {
229     _currentBufferedRows.reset()
230     _currentRawBytesWritten.reset()
231     _lastSerializedBytesWritten.reset()
232     _lastSerializeTime.reset()
233     currentShardNum = None
234     serializedBuffer.reset()
235     renewCompressedOutput()
236   }

238   def flush(force: Boolean, shardNum: Option[Int]): Unit =
239     if (force) {
240       doFlush(shardNum)
241     } else if (currentBufferedRows >= writeJob.writeOptions.batchSize) {
242       doFlush(shardNum)
243     }
244 
245   def doFlush(shardNum: Option[Int]): Unit = {
246     val client = nodeClient(shardNum)
247     val data = serialize()
248     var writeTime = 0L
249     Utils.retry[Unit, RetryableCHException](
250       writeJob.writeOptions.maxRetry,
251       writeJob.writeOptions.retryInterval
252     ) {
253       var startWriteTime = System.currentTimeMillis
254       client.syncInsertOutputJSONEachRow(database, table, format, codec, new ByteArrayInputStream(data)) match {
255         case Right(_) =>
256           writeTime = System.currentTimeMillis - startWriteTime
257           _totalWriteTime.add(writeTime)
258           _totalRecordsWritten.add(currentBufferedRows)
259         case Left(retryable) if writeJob.writeOptions.retryableErrorCodes.contains(retryable.code) =>
260           startWriteTime = System.currentTimeMillis
261           throw RetryableCHException(retryable.code, retryable.reason, Some(client.nodeSpec))
262         case Left(rethrow) => throw rethrow
263       }
264     } match {
265       case Success(_) =>
266         log.info(
267           s"""Job[${writeJob.queryId}]: batch write completed
268              |cluster: ${writeJob.cluster.map(_.name).getOrElse("none")}, shard: ${shardNum.getOrElse("none")}
269              |node: ${client.nodeSpec}
270              |        row count: $currentBufferedRows
271              |         raw size: ${Utils.bytesToString(currentBufferedRawBytes)}
272              |           format: $format
273              |compression codec: $codec
274              |  serialized size: ${Utils.bytesToString(lastSerializedBytesWritten)}
275              |   serialize time: ${lastSerializeTime}ms
276              |       write time: ${writeTime}ms
277              |""".stripMargin
278         )
279         reset()
280       case Failure(rethrow) => throw rethrow
281     }
282   }
283 
284   override def commit(): WriterCommitMessage = {
285     flush(currentBufferedRows > 0, currentShardNum)
286     CommitMessage(s"Job[${writeJob.queryId}]: commit")
287   }
288 
289   override def abort(): Unit = {}
290 
291   override def close(): Unit = {
292     IOUtils.closeQuietly(output)
293     client match {
294       case Left(clusterClient) => clusterClient.close()
295       case Right(nodeClient) => nodeClient.close()
296     }
297   }
