48   def buildNodeSpec(options: CaseInsensitiveStringMap): NodeSpec = {
49     val clientOpts = options.asScala
50       .filterKeys(_.startsWith(CATALOG_PROP_OPTION_PREFIX))
51       .map { case (k, v) => k.substring(CATALOG_PROP_OPTION_PREFIX.length) -> v }
52       .toMap
53       .filterKeys { key =>
54         val ignore = CATALOG_PROP_IGNORE_OPTIONS.contains(key)
55         if (ignore) {
56           log.warn(s"Ignore configuration $key.")
57         }
58         !ignore
59       }
60       .toMap
61     NodeSpec(
62       _host = options.getOrDefault(CATALOG_PROP_HOST, "localhost"),
63       _tcp_port = Some(options.getInt(CATALOG_PROP_TCP_PORT, 9000)),
64       _http_port = Some(options.getInt(CATALOG_PROP_HTTP_PORT, 8123)),
65       protocol = ClickHouseProtocol.fromUriScheme(options.getOrDefault(CATALOG_PROP_PROTOCOL, "http")),
66       username = options.getOrDefault(CATALOG_PROP_USER, "default"),
67       password = options.getOrDefault(CATALOG_PROP_PASSWORD, ""),
68       database = options.getOrDefault(CATALOG_PROP_DATABASE, "default"),
69       options = new JHashMap(clientOpts.asJava)
70     )
71   }

73   def queryClusterSpecs(nodeSpec: NodeSpec)(implicit nodeClient: NodeClient): Seq[ClusterSpec] = {
74     val clustersOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
75       """ SELECT
76         |   `cluster`,                 -- String
77         |   `shard_num`,               -- UInt32
78         |   `shard_weight`,            -- UInt32
79         |   `replica_num`,             -- UInt32
80         |   `host_name`,               -- String
81         |   `host_address`,            -- String
82         |   `port`,                    -- UInt16
83         |   `is_local`,                -- UInt8
84         |   `user`,                    -- String
85         |   `default_database`,        -- String
86         |   `errors_count`,            -- UInt32
87         |   `estimated_recovery_time`  -- UInt32
88         | FROM `system`.`clusters`
89         |""".stripMargin
90     )
91     clustersOutput.records
92       .groupBy(_.get("cluster").asText)
93       .map { case (cluster, rows) =>
94         val shards = rows
95           .groupBy(_.get("shard_num").asInt)
96           .map { case (shardNum, rows) =>
97             val shardWeight = rows.head.get("shard_weight").asInt
98             val nodes = rows.map { row =>
99               val replicaNum = row.get("replica_num").asInt
100               // should other properties be provided by `SparkConf`?
101               val clickhouseNode = nodeSpec.copy(
102                 // host_address is not works for testcontainers
103                 _host = row.get("host_name").asText,
104                 _tcp_port = Some(row.get("port").asInt),
105                 _http_port = if (Utils.isTesting) Some(8123) else nodeSpec.http_port
106               )
107               ReplicaSpec(replicaNum, clickhouseNode)
108             }.toArray
109             ShardSpec(shardNum, shardWeight, nodes)
110           }.toArray
111         ClusterSpec(cluster, shards)
112       }.toSeq
113   }

223     database: String,
224     table: String,

226   )(implicit nodeClient: NodeClient): StructType = {

228       s"""SELECT
229          |  `database`,                -- String
230          |  `table`,                   -- String
231          |  `name`,                    -- String
232          |  `type`,                    -- String

260   def queryPartitionSpec(
261     database: String,
262     table: String
263   )(implicit nodeClient: NodeClient): Seq[PartitionSpec] = {
264     val partOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
265       s"""SELECT
266          |  partition,                           -- String
267          |  partition_id,                        -- String
268          |  sum(rows)          AS row_count,     -- UInt64
269          |  sum(bytes_on_disk) AS size_in_bytes  -- UInt64
270          |FROM `system`.`parts`
271          |WHERE `database`='$database' AND `table`='$table' AND `active`=1
272          |GROUP BY `partition`, `partition_id`
273          |ORDER BY `partition` ASC, partition_id ASC
274          |""".stripMargin
275     )
276     if (partOutput.isEmpty || partOutput.rows == 1 && partOutput.records.head.get("partition").asText == "tuple()") {
277       return Array(NoPartitionSpec)
278     }
279     partOutput.records.map { row =>
280       PartitionSpec(
281         partition_value = row.get("partition").asText,
282         partition_id = row.get("partition_id").asText,
283         row_count = row.get("row_count").asLong,
284         size_in_bytes = row.get("size_in_bytes").asLong
285       )
286     }
287   }

299     database: String,
300     table: String,
301     partitionExpr: String,
302     cluster: Option[String] = None
303   )(implicit
304     nodeClient: NodeClient
305   ): Boolean =
306     nodeClient.syncQueryOutputJSONEachRow(
307       s"ALTER TABLE `$database`.`$table` ${cluster.map(c => s"ON CLUSTER $c").getOrElse("")} DROP PARTITION $partitionExpr"

309       case Right(_) => true

311         log.error(s"[${ex.code}]: ${ex.getMessage}")
