73   private def loadV2FunctionOpt(
74     name: String,
75     args: Seq[Expression],
76     functionRegistry: FunctionRegistry
77   ): Option[BoundFunction] = {
78     def loadFunction(ident: Identifier): UnboundFunction =
79       functionRegistry.load(ident.name).getOrElse(throw new NoSuchFunctionException(ident))
80     val inputType = StructType(args.zipWithIndex.map {
81       case (exp, pos) => StructField(s"_$pos", exp.dataType, exp.nullable)

83     try {
84       val unbound = loadFunction(Identifier.of(Array.empty, name))

93     }

108   private class CustomResolveTimeZone(timeZoneId: Option[String]) extends Rule[LogicalPlan] {
109     override def apply(plan: LogicalPlan): LogicalPlan =
110       plan.resolveExpressionsWithPruning(_.containsAnyPattern(LIST_SUBQUERY, TIME_ZONE_AWARE_EXPRESSION)) {
111         case e: TimeZoneAwareExpression if e.timeZoneId.isEmpty =>
112           e.withTimeZone(timeZoneId.getOrElse(conf.sessionLocalTimeZone))
113         // Casts could be added in the subquery plan through the rule TypeCoercion while coercing
114         // the types between the value expression and list query expression of IN expression.
115         // We need to subject the subquery plan through ResolveTimeZone again to setup timezone
116         // information for time zone aware expressions.
117         case e: ListQuery => e.withNewPlan(apply(e.plan))
118       }
119   }

140   def toCatalyst(
141     v2Expr: V2Expression,
142     fields: Array[StructField],
143     functionRegistry: FunctionRegistry
144   ): Expression =
145     v2Expr match {
146       case IdentityTransform(ref) => toCatalyst(ref, fields, functionRegistry)
147       case ref: NamedReference if ref.fieldNames.length == 1 =>
148         val (field, ordinal) = fields
149           .zipWithIndex
150           .find { case (field, _) => field.name == ref.fieldNames.head }
151           .getOrElse(throw CHClientException(s"Invalid field reference: $ref"))
152         BoundReference(ordinal, field.dataType, field.nullable)
153       case t: Transform =>
154         val catalystArgs = t.arguments().map(toCatalyst(_, fields, functionRegistry))
155         loadV2FunctionOpt(t.name(), catalystArgs, functionRegistry)
156           .map(bound => TransformExpression(bound, catalystArgs)).getOrElse {
157             throw CHClientException(s"Unsupported expression: $v2Expr")
158           }
159       case literal: LiteralValue[Any] => expressions.Literal(literal.value)
160       case _ => throw CHClientException(
161           s"Unsupported expression: $v2Expr"
162         )
163     }

165   def toSparkTransformOpt(expr: Expr, functionRegistry: FunctionRegistry): Option[Transform] =
166     Try(toSparkExpression(expr, functionRegistry)) match {
167       // need this function because spark `Table`'s `partitioning` field should be `Transform`
168       case Success(t: Transform) => Some(t)
169       case Success(_) => None
170       case Failure(_) if conf.getConf(IGNORE_UNSUPPORTED_TRANSFORM) => None
171       case Failure(rethrow) => throw new AnalysisException(rethrow.getMessage, cause = Some(rethrow))
172     }

174   def toSparkExpression(expr: Expr, functionRegistry: FunctionRegistry): V2Expression =
175     expr match {
176       case FieldRef(col) => identity(col)
177       case StringLiteral(value) => literal(value) // TODO LiteralTransform
178       case FuncExpr("rand", Nil) => apply("rand")
179       case FuncExpr("toYYYYMMDD", List(FuncExpr("toDate", List(FieldRef(col))))) => identity(col)
180       case FuncExpr(funName, args) if functionRegistry.clickHouseToSparkFunc.contains(funName) =>
181         apply(functionRegistry.clickHouseToSparkFunc(funName), args.map(toSparkExpression(_, functionRegistry)): _*)
182       case unsupported => throw CHClientException(s"Unsupported ClickHouse expression: $unsupported")
183     }

185   def toClickHouse(
186     transform: Transform,
187     functionRegistry: FunctionRegistry
188   ): Expr = transform match {
189     case IdentityTransform(fieldRefs) => FieldRef(fieldRefs.describe)
190     case ApplyTransform(name, args) if functionRegistry.sparkToClickHouseFunc.contains(name) =>
191       FuncExpr(functionRegistry.sparkToClickHouseFunc(name), args.map(arg => SQLExpr(arg.describe)).toList)
192     case bucket: BucketTransform => throw CHClientException(s"Bucket transform not support yet: $bucket")
193     case other: Transform => throw CHClientException(s"Unsupported transform: $other")
194   }

196   def inferTransformSchema(
197     primarySchema: StructType,
198     secondarySchema: StructType,
199     transform: Transform,
200     functionRegistry: FunctionRegistry
201   ): StructField = transform match {
202     case IdentityTransform(FieldReference(Seq(col))) => primarySchema.find(_.name == col)
203         .orElse(secondarySchema.find(_.name == col))
204         .getOrElse(throw CHClientException(s"Invalid partition column: $col"))
205     case t @ ApplyTransform(transformName, _) if functionRegistry.load(transformName).isDefined =>
206       val resType = functionRegistry.load(transformName) match {
207         case Some(f: ScalarFunction[_]) => f.resultType
208         case other => throw CHClientException(s"Unsupported function: $other")
209       }
210       StructField(t.toString, resType)
211     case bucket: BucketTransform => throw CHClientException(s"Bucket transform not support yet: $bucket")
212     case other: Transform => throw CHClientException(s"Unsupported transform: $other")
213   }
