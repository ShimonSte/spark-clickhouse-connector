26   def fromClickHouseType(chColumn: ClickHouseColumn): (DataType, Boolean) = {
27     val catalystType = chColumn.getDataType match {
28       case Nothing => NullType
29       case Bool => BooleanType
30       case String | JSON | UUID | Enum8 | Enum16 | IPv4 | IPv6 => StringType
31       case FixedString =>
32         conf.getConf(READ_FIXED_STRING_AS) match {
33           case "binary" => BinaryType
34           case "string" => StringType
35           case unsupported => throw CHClientException(s"Unsupported fixed string read format mapping: $unsupported")
36         }
37       case Int8 => ByteType
38       case UInt8 | Int16 => ShortType
39       case UInt16 | Int32 => IntegerType
40       case UInt32 | Int64 | UInt64 => LongType
41       case Int128 | UInt128 | Int256 | UInt256 => DecimalType(38, 0)
42       case Float32 => FloatType
43       case Float64 => DoubleType
44       case Date | Date32 => DateType
45       case DateTime | DateTime32 | DateTime64 => TimestampType
46       case ClickHouseDataType.Decimal if chColumn.getScale <= 38 =>
47         DecimalType(chColumn.getPrecision, chColumn.getScale)
48       case Decimal32 => DecimalType(9, chColumn.getScale)
49       case Decimal64 => DecimalType(18, chColumn.getScale)
50       case Decimal128 => DecimalType(38, chColumn.getScale)
51       case IntervalYear => YearMonthIntervalType(YearMonthIntervalType.YEAR)
52       case IntervalMonth => YearMonthIntervalType(YearMonthIntervalType.MONTH)
53       case IntervalDay => DayTimeIntervalType(DayTimeIntervalType.DAY)
54       case IntervalHour => DayTimeIntervalType(DayTimeIntervalType.HOUR)
55       case IntervalMinute => DayTimeIntervalType(DayTimeIntervalType.MINUTE)
56       case IntervalSecond => DayTimeIntervalType(DayTimeIntervalType.SECOND)
57       case Array =>
58         val elementChCols = chColumn.getNestedColumns
59         assert(elementChCols.size == 1)
60         val (elementType, elementNullable) = fromClickHouseType(elementChCols.get(0))
61         ArrayType(elementType, elementNullable)
62       case Map =>
63         val kvChCols = chColumn.getNestedColumns
64         assert(kvChCols.size == 2)
65         val (keyChType, valueChType) = (kvChCols.get(0), kvChCols.get(1))
66         val (keyType, keyNullable) = fromClickHouseType(keyChType)
67         require(
68           !keyNullable,
69           s"Illegal type: ${keyChType.getOriginalTypeName}, the key type of Map should not be nullable"
70         )
71         val (valueType, valueNullable) = fromClickHouseType(valueChType)
72         MapType(keyType, valueType, valueNullable)
73       case Object | Nested | Tuple | Point | Polygon | MultiPolygon | Ring | IntervalQuarter | IntervalWeek |
74           Decimal256 | AggregateFunction | SimpleAggregateFunction =>
75         throw CHClientException(s"Unsupported type: ${chColumn.getOriginalTypeName}")
76     }
77     (catalystType, chColumn.isNullable)
78   }

80   def toClickHouseType(catalystType: DataType, nullable: Boolean): String =
81     catalystType match {
82       case BooleanType => maybeNullable("UInt8", nullable)
83       case ByteType => maybeNullable("Int8", nullable)
84       case ShortType => maybeNullable("Int16", nullable)
85       case IntegerType => maybeNullable("Int32", nullable)
86       case LongType => maybeNullable("Int64", nullable)
87       case FloatType => maybeNullable("Float32", nullable)
88       case DoubleType => maybeNullable("Float64", nullable)
89       case StringType => maybeNullable("String", nullable)
90       case VarcharType(_) => maybeNullable("String", nullable)
91       case CharType(_) => maybeNullable("String", nullable) // TODO: maybe FixString?
92       case DateType => maybeNullable("Date", nullable)
93       case TimestampType => maybeNullable("DateTime", nullable)
94       case DecimalType.Fixed(p, s) => maybeNullable(s"Decimal($p, $s)", nullable)
95       case ArrayType(elemType, containsNull) => s"Array(${toClickHouseType(elemType, containsNull)})"
96       // TODO currently only support String as key
97       case MapType(keyType, valueType, valueContainsNull) if keyType.isInstanceOf[StringType] =>
98         s"Map(${toClickHouseType(keyType, nullable = false)}, ${toClickHouseType(valueType, valueContainsNull)})"
99       case _ => throw CHClientException(s"Unsupported type: $catalystType")
100     }

102   def fromClickHouseSchema(chSchema: Seq[(String, String)]): StructType = {
103     val structFields = chSchema.map { case (name, maybeNullableType) =>
104       val chCols = ClickHouseColumn.parse(s"`$name` $maybeNullableType")
105       assert(chCols.size == 1)
106       val (sparkType, nullable) = fromClickHouseType(chCols.get(0))
107       StructField(name, sparkType, nullable)
108     }
109     StructType(structFields)
110   }
111 
112   def toClickHouseSchema(catalystSchema: StructType): Seq[(String, String, String)] =
113     catalystSchema.fields
114       .map { field =>
115         val chType = toClickHouseType(field.dataType, field.nullable)
116         (field.name, chType, field.getComment().map(c => s" COMMENT '$c'").getOrElse(""))
117       }

119   private[clickhouse] def maybeNullable(chType: String, nullable: Boolean): String =
120     if (nullable) wrapNullable(chType) else chType
