59   protected lazy val shardExpr: Option[Expression] = writeJob.sparkShardExpr match {
60     case None => None
61     case Some(v2Expr) =>
62       val catalystExpr = ExprUtils.toCatalyst(v2Expr, writeJob.dataSetSchema.fields, writeJob.functionRegistry)
63       catalystExpr match {
64         case BoundReference(_, dataType, _)
65             if dataType.isInstanceOf[ByteType] // list all integral types here because we can not access `IntegralType`
66               || dataType.isInstanceOf[ShortType]
67               || dataType.isInstanceOf[IntegerType]
68               || dataType.isInstanceOf[LongType] =>
69           Some(catalystExpr)
70         case BoundReference(_, dataType, _) =>
71           throw CHClientException(s"Invalid data type of sharding field: $dataType")
72         case TransformExpression(function, _, _) =>
73           function.resultType match {
74             case ByteType | ShortType | IntegerType | LongType => Some(catalystExpr)
75             case _ => throw CHClientException(s"Invalid data type of sharding field: ${function.resultType}")
76           }
77         case unsupported: Expression =>
78           log.warn(s"Unsupported expression of sharding field: $unsupported")
79           None
80       }
81   }
82 
83   protected lazy val shardProjection: Option[expressions.Projection] = shardExpr
84     .filter(_ => writeJob.writeOptions.convertDistributedToLocal)
85     .flatMap {
86       case expr: BoundReference =>
87         Some(SafeProjection.create(Seq(expr)))
88       case expr @ TransformExpression(function, _, _) =>
89         // result type must be integer class
90         function.resultType match {
91           case ByteType => classOf[Byte]
92           case ShortType => classOf[Short]
93           case IntegerType => classOf[Int]
94           case LongType => classOf[Long]
95           case _ => throw CHClientException(s"Invalid return data type for function ${function.name()}," +
96               s"sharding field: ${function.resultType}")
97         }
98         Some(SafeProjection.create(Seq(ExprUtils.resolveTransformCatalyst(expr, Some(writeJob.tz.getId)))))
99     }

101   // put the node select strategy in executor side because we need to calculate shard and don't know the records
102   // util DataWriter#write(InternalRow) invoked.
103   protected lazy val client: Either[ClusterClient, NodeClient] =
104     writeJob.tableEngineSpec match {
105       case _: DistributedEngineSpec
106           if writeJob.writeOptions.useClusterNodesForDistributed || writeJob.writeOptions.convertDistributedToLocal =>
107         val clusterSpec = writeJob.cluster.get
108         log.info(s"Connect to cluster ${clusterSpec.name}, which has ${clusterSpec.shards.length} shards and " +
109           s"${clusterSpec.nodes.length} nodes.")
110         Left(ClusterClient(clusterSpec))
111       case _ =>
112         val nodeSpec = writeJob.node
113         log.info(s"Connect to single node: $nodeSpec")
114         Right(NodeClient(nodeSpec))
115     }

117   def nodeClient(shardNum: Option[Int]): NodeClient = client match {

122   def calcShard(record: InternalRow): Option[Int] = (shardExpr, shardProjection) match {
123     case (Some(BoundReference(_, dataType, _)), Some(projection)) =>
124       doCalcShard(record, dataType, projection)
125     case (Some(TransformExpression(function, _, _)), Some(projection)) =>
126       doCalcShard(record, function.resultType, projection)
127     case _ => None
128   }
129 
130   private def doCalcShard(record: InternalRow, dataType: DataType, projection: Projection): Option[Int] = {
131     val shardValue = dataType match {
132       case ByteType => Some(projection(record).getByte(0).toLong)
133       case ShortType => Some(projection(record).getShort(0).toLong)
134       case IntegerType => Some(projection(record).getInt(0).toLong)
135       case LongType => Some(projection(record).getLong(0))
136       case _ => None
137     }
138     shardValue.map(value => ShardUtils.calcShard(writeJob.cluster.get, value).num)
139   }

182   private def renewCompressedOutput(): Unit = {
183     val compressedOutput = (codec, protocol) match {
184       case (ClickHouseCompression.NONE, _) => observableSerializedOutput
185       case (ClickHouseCompression.LZ4, ClickHouseProtocol.HTTP) =>
186         // clickhouse http client forces compressed output stream
187         // new Lz4OutputStream(observableSerializedOutput, 4 * 1024 * 1024, null)
188         observableSerializedOutput
189       case unsupported =>
190         throw CHClientException(s"unsupported compression codec: $unsupported")
191     }
192     compressedForwardingOutput.updateDelegate(compressedOutput)
193   }

208   override def write(record: InternalRow): Unit = {
209     val shardNum = calcShard(record)
210     flush(force = shardNum != currentShardNum && currentBufferedRows > 0, currentShardNum)
211     currentShardNum = shardNum
212     val (_, serializedTime) = Utils.timeTakenMs(writeRow(record))
213     _lastSerializeTime.add(serializedTime)
214     _totalSerializeTime.add(serializedTime)
215     _currentBufferedRows.add(1)
216     flush(force = false, currentShardNum)
217   }

221   def serialize(): Array[Byte] = {
222     val (data, serializedTime) = Utils.timeTakenMs(doSerialize())
223     _lastSerializeTime.add(serializedTime)
224     _totalSerializeTime.add(serializedTime)

230   def reset(): Unit = {
231     _currentBufferedRows.reset()
232     _currentRawBytesWritten.reset()
233     _lastSerializedBytesWritten.reset()
234     _lastSerializeTime.reset()
235     currentShardNum = None

240   def flush(force: Boolean, shardNum: Option[Int]): Unit =
241     if (force) {
242       doFlush(shardNum)
243     } else if (currentBufferedRows >= writeJob.writeOptions.batchSize) {
244       doFlush(shardNum)
245     }
246 
247   def doFlush(shardNum: Option[Int]): Unit = {
248     val client = nodeClient(shardNum)
249     val data = serialize()
250     var writeTime = 0L
251     Utils.retry[Unit, RetryableCHException](
252       writeJob.writeOptions.maxRetry,
253       writeJob.writeOptions.retryInterval
254     ) {
255       var startWriteTime = System.currentTimeMillis
256       client.syncInsertOutputJSONEachRow(database, table, format, codec, new ByteArrayInputStream(data)) match {
257         case Right(_) =>
258           writeTime = System.currentTimeMillis - startWriteTime
259           _totalWriteTime.add(writeTime)
260           _totalRecordsWritten.add(currentBufferedRows)
261         case Left(retryable) if writeJob.writeOptions.retryableErrorCodes.contains(retryable.code) =>
262           startWriteTime = System.currentTimeMillis
263           throw RetryableCHException(retryable.code, retryable.reason, Some(client.nodeSpec))
264         case Left(rethrow) => throw rethrow
265       }
266     } match {
267       case Success(_) =>
268         log.info(
269           s"""Job[${writeJob.queryId}]: batch write completed
270              |cluster: ${writeJob.cluster.map(_.name).getOrElse("none")}, shard: ${shardNum.getOrElse("none")}
271              |node: ${client.nodeSpec}
272              |        row count: $currentBufferedRows
273              |         raw size: ${Utils.bytesToString(currentBufferedRawBytes)}
274              |           format: $format
275              |compression codec: $codec
276              |  serialized size: ${Utils.bytesToString(lastSerializedBytesWritten)}
277              |   serialize time: ${lastSerializeTime}ms
278              |       write time: ${writeTime}ms
279              |""".stripMargin
280         )
281         reset()
282       case Failure(rethrow) => throw rethrow
283     }
284   }
285 
286   override def commit(): WriterCommitMessage = {
287     flush(currentBufferedRows > 0, currentShardNum)
288     CommitMessage(s"Job[${writeJob.queryId}]: commit")
289   }
290 
291   override def abort(): Unit = {}
292 
293   override def close(): Unit = {
294     IOUtils.closeQuietly(output)
295     client match {
296       case Left(clusterClient) => clusterClient.close()
297       case Right(nodeClient) => nodeClient.close()
298     }
299   }
