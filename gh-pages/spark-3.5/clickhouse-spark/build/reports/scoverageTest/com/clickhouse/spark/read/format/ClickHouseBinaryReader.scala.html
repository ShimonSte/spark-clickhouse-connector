50   override def decode(record: ClickHouseRecord): InternalRow = {
51     val values: Array[Any] = new Array[Any](record.size)
52     if (readSchema.nonEmpty) {
53       var i: Int = 0
54       while (i < record.size) {
55         values(i) = decodeValue(record.getValue(i), readSchema.fields(i))
56         i = i + 1
57       }
58     }
59     new GenericInternalRow(values)
60   }

62   private def decodeValue(value: ClickHouseValue, structField: StructField): Any = {
63     if (value == null || value.isNullOrEmpty && value.isNullable) {
64       // should we check `structField.nullable`?
65       return null
66     }
67 
68     structField.dataType match {
69       case BooleanType => value.asBoolean
70       case ByteType => value.asByte
71       case ShortType => value.asShort
72       case IntegerType => value.asInteger
73       case LongType => value.asLong
74       case FloatType => value.asFloat
75       case DoubleType => value.asDouble
76       case d: DecimalType => Decimal(value.asBigDecimal(d.scale))
77       case TimestampType =>
78         var _instant = value.asZonedDateTime.withZoneSameInstant(ZoneOffset.UTC)
79         TimeUnit.SECONDS.toMicros(_instant.toEpochSecond) + TimeUnit.NANOSECONDS.toMicros(_instant.getNano())
80       case StringType if value.isInstanceOf[ClickHouseStringValue] => UTF8String.fromBytes(value.asBinary)
81       case StringType => UTF8String.fromString(value.asString)
82       case DateType => value.asDate.toEpochDay.toInt
83       case BinaryType => value.asBinary
84       case ArrayType(_dataType, _nullable) =>
85         val arrayValue = value.asInstanceOf[ClickHouseArraySequence]
86         val convertedArray = Array.tabulate(arrayValue.length) { i =>
87           decodeValue(
88             arrayValue.getValue(i, createClickHouseValue(null, _dataType)),
89             StructField("element", _dataType, _nullable)
90           )
91         }
92         new GenericArrayData(convertedArray)
93       case MapType(_keyType, _valueType, _valueNullable) =>
94         val convertedMap = value.asMap().asScala.map { case (rawKey, rawValue) =>
95           val decodedKey = decodeValue(createClickHouseValue(rawKey, _keyType), StructField("key", _keyType, false))
96           val decodedValue =
97             decodeValue(createClickHouseValue(rawValue, _valueType), StructField("value", _valueType, _valueNullable))
98           (decodedKey, decodedValue)
99         }
100         ArrayBasedMapData(convertedMap)
101 
102       case _ =>
103         throw CHClientException(s"Unsupported catalyst type ${structField.name}[${structField.dataType}]")
104     }
105   }
106 
107   private def createClickHouseValue(rawValue: Any, dataType: DataType): ClickHouseValue = {
108     val isNull = rawValue == null
109 
110     dataType match {
111       case StringType =>
112         if (isNull) ClickHouseStringValue.ofNull()
113         else ClickHouseStringValue.of(rawValue.toString)
114 
115       case IntegerType =>
116         if (isNull) ClickHouseIntegerValue.ofNull()
117         else ClickHouseIntegerValue.of(rawValue.asInstanceOf[Int])
118 
119       case LongType =>
120         if (isNull) ClickHouseLongValue.ofNull()
121         else ClickHouseLongValue.of(rawValue.asInstanceOf[Long])
122 
123       case DoubleType =>
124         if (isNull) ClickHouseDoubleValue.ofNull()
125         else ClickHouseDoubleValue.of(rawValue.asInstanceOf[Double])
126 
127       case FloatType =>
128         if (isNull) ClickHouseFloatValue.ofNull()
129         else ClickHouseFloatValue.of(rawValue.asInstanceOf[Float])
130 
131       case BooleanType =>
132         if (isNull) ClickHouseBoolValue.ofNull()
133         else ClickHouseBoolValue.of(rawValue.asInstanceOf[Boolean])
134 
135       case _: ArrayType =>
136         if (isNull) ClickHouseArrayValue.ofEmpty()
137         else ClickHouseArrayValue.of(rawValue.asInstanceOf[Array[Object]])
138 
139       case _: MapType =>
140         if (isNull) ClickHouseMapValue.ofEmpty(classOf[Object], classOf[Object])
141         else ClickHouseMapValue.of(
142           rawValue.asInstanceOf[java.util.Map[Object, Object]],
143           classOf[Object],
144           classOf[Object]
145         )
146 
147       case _ =>
148         if (isNull) ClickHouseStringValue.ofNull()
149         else ClickHouseStringValue.of(rawValue.toString)
150     }
151   }
