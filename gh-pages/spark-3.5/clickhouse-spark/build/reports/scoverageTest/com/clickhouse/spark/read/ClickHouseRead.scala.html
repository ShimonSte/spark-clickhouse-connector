37 class ClickHouseScanBuilder(
38   scanJob: ScanJobDescription,
39   physicalSchema: StructType,
40   metadataSchema: StructType,
41   partitionTransforms: Array[Transform]
42 ) extends ScanBuilder
43     with SupportsPushDownLimit
44     with SupportsPushDownFilters
45     with SupportsPushDownAggregates
46     with SupportsPushDownRequiredColumns
47     with ClickHouseHelper
48     with SQLHelper
49     with Logging {
50 
51   implicit private val tz: ZoneId = scanJob.tz
52 
53   private val reservedMetadataSchema: StructType = StructType(
54     metadataSchema.dropWhile(field => physicalSchema.fields.map(_.name).contains(field.name))
55   )
56 
57   private var _readSchema: StructType = StructType(
58     physicalSchema.fields ++ reservedMetadataSchema.fields
59   )
60 
61   private var _limit: Option[Int] = None
62 
63   override def pushLimit(limit: Int): Boolean = {
64     this._limit = Some(limit)
65     true
66   }
67 
68   private var _pushedFilters = Array.empty[Filter]
69 
70   override def pushedFilters: Array[Filter] = this._pushedFilters
71 
72   override def pushFilters(filters: Array[Filter]): Array[Filter] = {
73     val (pushed, unSupported) = filters.partition(f => compileFilter(f).isDefined)
74     this._pushedFilters = pushed
75     unSupported
76   }
77 
78   private var _pushedGroupByCols: Option[Array[String]] = None
79   private var _groupByClause: Option[String] = None
80 
81   override def pushAggregation(aggregation: Aggregation): Boolean = {
82     val compiledAggs = aggregation.aggregateExpressions.flatMap(compileAggregate)
83     if (compiledAggs.length != aggregation.aggregateExpressions.length) return false
84 
85     val compiledGroupByCols = aggregation.groupByExpressions.map(_.toString)
86 
87     // The column names here are already quoted and can be used to build sql string directly.
88     // e.g. [`DEPT`, `NAME`, MAX(`SALARY`), MIN(`BONUS`)] =>
89     //        SELECT `DEPT`, `NAME`, MAX(`SALARY`), MIN(`BONUS`)
90     //        FROM `test`.`employee`
91     //        WHERE 1=0
92     //        GROUP BY `DEPT`, `NAME`
93     val compiledSelectItems = compiledGroupByCols ++ compiledAggs
94     val groupByClause = if (compiledGroupByCols.nonEmpty) "GROUP BY " + compiledGroupByCols.mkString(", ") else ""
95     val aggQuery =
96       s"""SELECT ${compiledSelectItems.mkString(", ")}
97          |FROM ${quoted(scanJob.tableSpec.database)}.${quoted(scanJob.tableSpec.name)}
98          |WHERE 1=0
99          |$groupByClause
100          |""".stripMargin
101     try {
102       _readSchema = Utils.tryWithResource(NodeClient(scanJob.node)) { implicit nodeClient: NodeClient =>
103         val fields = (getQueryOutputSchema(aggQuery) zip compiledSelectItems)
104           .map { case (structField, colExpr) => structField.copy(name = colExpr) }
105         StructType(fields)
106       }
107       _pushedGroupByCols = Some(compiledGroupByCols)
108       _groupByClause = Some(groupByClause)
109       true
110     } catch {
111       case NonFatal(e) =>
112         log.error("Failed to push down aggregation to ClickHouse", e)
113         false
114     }
115   }
116 
117   override def pruneColumns(requiredSchema: StructType): Unit = {
118     val requiredCols = requiredSchema.map(_.name)
119     this._readSchema = StructType(_readSchema.filter(field => requiredCols.contains(field.name)))
120   }
121 
122   override def build(): Scan = new ClickHouseBatchScan(scanJob.copy(
123     readSchema = _readSchema,
124     filtersExpr = compileFilters(AlwaysTrue :: pushedFilters.toList),
125     groupByClause = _groupByClause,
126     limit = _limit
127   ))
128 }

144   lazy val inputPartitions: Array[ClickHouseInputPartition] = scanJob.tableEngineSpec match {
145     case DistributedEngineSpec(_, _, local_db, local_table, _, _) if scanJob.readOptions.convertDistributedToLocal =>
146       scanJob.cluster.get.shards.flatMap { shardSpec =>
147         Utils.tryWithResource(NodeClient(shardSpec.nodes.head)) { implicit nodeClient: NodeClient =>
148           queryPartitionSpec(local_db, local_table).map { partitionSpec =>
149             ClickHouseInputPartition(
150               scanJob.localTableSpec.get,
151               partitionSpec,
152               scanJob.readOptions.splitByPartitionId,
153               shardSpec // TODO pickup preferred
154             )
155           }
156         }
157       }
158     case _: DistributedEngineSpec if scanJob.readOptions.useClusterNodesForDistributed =>
159       throw CHClientException(
160         s"${READ_DISTRIBUTED_USE_CLUSTER_NODES.key} is not supported yet."
161       )
162     case _: DistributedEngineSpec =>
163       // we can not collect all partitions from single node, thus should treat table as no partitioned table
164       Array(ClickHouseInputPartition(
165         scanJob.tableSpec,
166         NoPartitionSpec,
167         scanJob.readOptions.splitByPartitionId,
168         scanJob.node
169       ))
170     case _: TableEngineSpec =>
171       Utils.tryWithResource(NodeClient(scanJob.node)) { implicit nodeClient: NodeClient =>
172         queryPartitionSpec(database, table).map { partitionSpec =>
173           ClickHouseInputPartition(
174             scanJob.tableSpec,
175             partitionSpec,
176             scanJob.readOptions.splitByPartitionId,
177             scanJob.node // TODO pickup preferred
178           )
179         }
180       }.toArray
181   }
182 
183   override def toBatch: Batch = this
184 
185   // may contains meta columns
186   override def readSchema(): StructType = scanJob.readSchema
187 
188   override def planInputPartitions: Array[InputPartition] = inputPartitions.toArray

190   // TODO KeyGroupedPartitioning

192 
193   override def createReaderFactory: PartitionReaderFactory = this

195   override def createReader(_partition: InputPartition): PartitionReader[InternalRow] = {
196     val format = scanJob.readOptions.format
197     val partition = _partition.asInstanceOf[ClickHouseInputPartition]
198     val finalScanJob = scanJob.copy(filtersExpr =
199       scanJob.filtersExpr + " AND "
200         + compileFilters(AlwaysTrue :: runtimeFilters.toList)
201     )
202     format match {
203       case "json" => new ClickHouseJsonReader(finalScanJob, partition)
204       case "binary" => new ClickHouseBinaryReader(finalScanJob, partition)
205       case unsupported => throw CHClientException(s"Unsupported read format: $unsupported")
206     }
207   }
208 
209   override def supportedCustomMetrics(): Array[CustomMetric] = Array(
210     BlocksReadMetric(),
211     BytesReadMetric()
212   )
213 
214   override def filterAttributes(): Array[NamedReference] =
215     if (scanJob.readOptions.runtimeFilterEnabled) {
216       scanJob.readSchema.fields.map(field => Expressions.column(field.name))
217     } else {
218       Array.empty
219     }
220 
221   override def filter(filters: Array[Filter]): Unit =
222     runtimeFilters = filters
