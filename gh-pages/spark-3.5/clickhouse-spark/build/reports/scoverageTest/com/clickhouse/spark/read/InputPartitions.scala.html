28 case class ClickHouseInputPartition(
29   table: TableSpec,
30   partition: PartitionSpec,
31   filterByPartitionId: Boolean,
32   candidateNodes: Nodes, // try to use them only when preferredNode unavailable
33   preferredNode: Option[NodeSpec] = None // TODO assigned by ScanBuilder in Spark Driver side
34 ) extends InputPartition {
35 
36   override def preferredLocations(): Array[String] = preferredNode match {
37     case Some(preferred) => Array(preferred.host)
38     case None => candidateNodes.nodes.map(_.host)
39   }
40 
41   def partFilterExpr: String = partition match {
42     case NoPartitionSpec => "1=1"
43     case PartitionSpec(_, partitionId, _, _) if filterByPartitionId =>
44       s"_partition_id = '$partitionId'"
45     case PartitionSpec(partitionValue, _, _, _) =>
46       s"${table.partition_key} = ${compilePartitionFilterValue(partitionValue)}"
47   }
48 
49   // TODO improve and test
50   def compilePartitionFilterValue(partitionValue: String): String =
51     (partitionValue.contains("-"), partitionValue.contains("(")) match {
52       // quote when partition by a single Date Type column to avoid illegal types of arguments (Date, Int64)
53       case (true, false) => s"'$partitionValue'"
54       // Date type column is quoted if there are multi partition columns
55       case _ => s"$partitionValue"
56     }
57 }
