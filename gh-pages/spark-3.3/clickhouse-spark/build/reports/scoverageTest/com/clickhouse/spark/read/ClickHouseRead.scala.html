35 class ClickHouseScanBuilder(
36   scanJob: ScanJobDescription,
37   physicalSchema: StructType,
38   metadataSchema: StructType,
39   partitionTransforms: Array[Transform]
40 ) extends ScanBuilder
41     with SupportsPushDownLimit
42     with SupportsPushDownFilters
43     with SupportsPushDownAggregates
44     with SupportsPushDownRequiredColumns
45     with ClickHouseHelper
46     with SQLHelper
47     with Logging {
48 
49   implicit private val tz: ZoneId = scanJob.tz
50 
51   private val reservedMetadataSchema: StructType = StructType(
52     metadataSchema.dropWhile(field => physicalSchema.fields.map(_.name).contains(field.name))
53   )
54 
55   private var _readSchema: StructType = StructType(
56     physicalSchema.fields ++ reservedMetadataSchema.fields
57   )
58 
59   private var _limit: Option[Int] = None
60 
61   override def pushLimit(limit: Int): Boolean = {
62     this._limit = Some(limit)
63     true
64   }
65 
66   private var _pushedFilters = Array.empty[Filter]
67 
68   override def pushedFilters: Array[Filter] = this._pushedFilters
69 
70   override def pushFilters(filters: Array[Filter]): Array[Filter] = {
71     val (pushed, unSupported) = filters.partition(f => compileFilter(f).isDefined)
72     this._pushedFilters = pushed
73     unSupported
74   }
75 
76   private var _pushedGroupByCols: Option[Array[String]] = None
77   private var _groupByClause: Option[String] = None
78 
79   override def pushAggregation(aggregation: Aggregation): Boolean = {
80     val compiledAggs = aggregation.aggregateExpressions.flatMap(compileAggregate)
81     if (compiledAggs.length != aggregation.aggregateExpressions.length) return false
82 
83     val compiledGroupByCols = aggregation.groupByExpressions.map(_.toString)
84 
85     // The column names here are already quoted and can be used to build sql string directly.
86     // e.g. [`DEPT`, `NAME`, MAX(`SALARY`), MIN(`BONUS`)] =>
87     //        SELECT `DEPT`, `NAME`, MAX(`SALARY`), MIN(`BONUS`)
88     //        FROM `test`.`employee`
89     //        WHERE 1=0
90     //        GROUP BY `DEPT`, `NAME`
91     val compiledSelectItems = compiledGroupByCols ++ compiledAggs
92     val groupByClause = if (compiledGroupByCols.nonEmpty) "GROUP BY " + compiledGroupByCols.mkString(", ") else ""
93     val aggQuery =
94       s"""SELECT ${compiledSelectItems.mkString(", ")}
95          |FROM ${quoted(scanJob.tableSpec.database)}.${quoted(scanJob.tableSpec.name)}
96          |WHERE 1=0
97          |$groupByClause
98          |""".stripMargin
99     try {
100       _readSchema = Utils.tryWithResource(NodeClient(scanJob.node)) { implicit nodeClient: NodeClient =>
101         val fields = (getQueryOutputSchema(aggQuery) zip compiledSelectItems)
102           .map { case (structField, colExpr) => structField.copy(name = colExpr) }
103         StructType(fields)
104       }
105       _pushedGroupByCols = Some(compiledGroupByCols)
106       _groupByClause = Some(groupByClause)
107       true
108     } catch {
109       case NonFatal(e) =>
110         log.error("Failed to push down aggregation to ClickHouse", e)
111         false
112     }
113   }
114 
115   override def pruneColumns(requiredSchema: StructType): Unit = {
116     val requiredCols = requiredSchema.map(_.name)
117     this._readSchema = StructType(_readSchema.filter(field => requiredCols.contains(field.name)))
118   }
119 
120   override def build(): Scan = new ClickHouseBatchScan(scanJob.copy(
121     readSchema = _readSchema,
122     filtersExpr = compileFilters(AlwaysTrue :: pushedFilters.toList),
123     groupByClause = _groupByClause,
124     limit = _limit
125   ))
126 }

142   lazy val inputPartitions: Array[ClickHouseInputPartition] = scanJob.tableEngineSpec match {
143     case DistributedEngineSpec(_, _, local_db, local_table, _, _) if scanJob.readOptions.convertDistributedToLocal =>
144       scanJob.cluster.get.shards.flatMap { shardSpec =>
145         Utils.tryWithResource(NodeClient(shardSpec.nodes.head)) { implicit nodeClient: NodeClient =>
146           queryPartitionSpec(local_db, local_table).map { partitionSpec =>
147             ClickHouseInputPartition(
148               scanJob.localTableSpec.get,

153           }
154         }

157       throw CHClientException(
158         s"${READ_DISTRIBUTED_USE_CLUSTER_NODES.key} is not supported yet."
159       )
160     case _: DistributedEngineSpec =>
161       // we can not collect all partitions from single node, thus should treat table as no partitioned table
162       Array(ClickHouseInputPartition(
163         scanJob.tableSpec,
164         NoPartitionSpec,
165         scanJob.readOptions.splitByPartitionId,
166         scanJob.node
167       ))
168     case _: TableEngineSpec =>
169       Utils.tryWithResource(NodeClient(scanJob.node)) { implicit nodeClient: NodeClient =>
170         queryPartitionSpec(database, table).map { partitionSpec =>
171           ClickHouseInputPartition(
172             scanJob.tableSpec,
173             partitionSpec,
174             scanJob.readOptions.splitByPartitionId,
175             scanJob.node // TODO pickup preferred
176           )
177         }
178       }.toArray
179   }

191   override def createReaderFactory: PartitionReaderFactory = this

193   override def createReader(_partition: InputPartition): PartitionReader[InternalRow] = {
194     val format = scanJob.readOptions.format
195     val partition = _partition.asInstanceOf[ClickHouseInputPartition]
196     val finalScanJob = scanJob.copy(filtersExpr =
197       scanJob.filtersExpr + " AND "
198         + compileFilters(AlwaysTrue :: runtimeFilters.toList)
199     )
200     format match {
201       case "json" => new ClickHouseJsonReader(finalScanJob, partition)
202       case "binary" => new ClickHouseBinaryReader(finalScanJob, partition)
203       case unsupported => throw CHClientException(s"Unsupported read format: $unsupported")
204     }
205   }
206 
207   override def supportedCustomMetrics(): Array[CustomMetric] = Array(
208     BlocksReadMetric(),
209     BytesReadMetric()
210   )
211 
212   override def filterAttributes(): Array[NamedReference] =
213     if (scanJob.readOptions.runtimeFilterEnabled) {
214       scanJob.readSchema.fields.map(field => Expressions.column(field.name))
215     } else {
216       Array.empty
217     }
218 
219   override def filter(filters: Array[Filter]): Unit =
220     runtimeFilters = filters
