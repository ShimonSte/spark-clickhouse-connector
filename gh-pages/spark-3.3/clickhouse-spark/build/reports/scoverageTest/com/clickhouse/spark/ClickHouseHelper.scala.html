48   def buildNodeSpec(options: CaseInsensitiveStringMap): NodeSpec = {
49     val clientOpts = options.asScala
50       .filterKeys(_.startsWith(CATALOG_PROP_OPTION_PREFIX))
51       .map { case (k, v) => k.substring(CATALOG_PROP_OPTION_PREFIX.length) -> v }
52       .toMap
53       .filterKeys { key =>
54         val ignore = CATALOG_PROP_IGNORE_OPTIONS.contains(key)
55         if (ignore) {
56           log.warn(s"Ignore configuration $key.")
57         }
58         !ignore
59       }.toMap
60     NodeSpec(
61       _host = options.getOrDefault(CATALOG_PROP_HOST, "localhost"),
62       _tcp_port = Some(options.getInt(CATALOG_PROP_TCP_PORT, 9000)),
63       _http_port = Some(options.getInt(CATALOG_PROP_HTTP_PORT, 8123)),
64       protocol = ClickHouseProtocol.fromUriScheme(options.getOrDefault(CATALOG_PROP_PROTOCOL, "http")),
65       username = options.getOrDefault(CATALOG_PROP_USER, "default"),
66       password = options.getOrDefault(CATALOG_PROP_PASSWORD, ""),
67       database = options.getOrDefault(CATALOG_PROP_DATABASE, "default"),
68       options = new JHashMap(clientOpts.asJava)
69     )
70   }

72   def queryClusterSpecs(nodeSpec: NodeSpec)(implicit nodeClient: NodeClient): Seq[ClusterSpec] = {
73     val clustersOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
74       """ SELECT
75         |   `cluster`,                 -- String
76         |   `shard_num`,               -- UInt32
77         |   `shard_weight`,            -- UInt32
78         |   `replica_num`,             -- UInt32
79         |   `host_name`,               -- String
80         |   `host_address`,            -- String
81         |   `port`,                    -- UInt16
82         |   `is_local`,                -- UInt8
83         |   `user`,                    -- String
84         |   `default_database`,        -- String
85         |   `errors_count`,            -- UInt32
86         |   `estimated_recovery_time`  -- UInt32
87         | FROM `system`.`clusters`
88         |""".stripMargin
89     )
90     clustersOutput.records
91       .groupBy(_.get("cluster").asText)
92       .map { case (cluster, rows) =>
93         val shards = rows
94           .groupBy(_.get("shard_num").asInt)
95           .map { case (shardNum, rows) =>
96             val shardWeight = rows.head.get("shard_weight").asInt
97             val nodes = rows.map { row =>
98               val replicaNum = row.get("replica_num").asInt
99               // should other properties be provided by `SparkConf`?
100               val clickhouseNode = nodeSpec.copy(
101                 // host_address is not works for testcontainers
102                 _host = row.get("host_name").asText,
103                 _tcp_port = Some(row.get("port").asInt),
104                 _http_port = if (Utils.isTesting) Some(8123) else nodeSpec.http_port
105               )
106               ReplicaSpec(replicaNum, clickhouseNode)
107             }.toArray
108             ShardSpec(shardNum, shardWeight, nodes)
109           }.toArray
110         ClusterSpec(cluster, shards)
111       }.toSeq
112   }

156          |  `is_temporary`,               -- UInt8

192       ),
193       dependencies_database = tableRow.get("dependencies_database").elements().asScala.map(_.asText).toList,
194       dependencies_table = tableRow.get("dependencies_table").elements().asScala.map(_.asText).toList,
195       create_table_query = tableRow.get("create_table_query").asText,
196       engine_full = tableRow.get("engine_full").asText,
197       partition_key = tableRow.get("partition_key").asText,
198       sorting_key = tableRow.get("sorting_key").asText,
199       primary_key = tableRow.get("primary_key").asText,
200       sampling_key = tableRow.get("sampling_key").asText,
201       storage_policy = tableRow.get("storage_policy").asText,
202       total_rows = tableRow.get("total_rows") match {
203         case _: NullNode | null => None
204         case node: JsonNode => Some(node.asLong)
205       },
206       total_bytes = tableRow.get("total_bytes") match {
207         case _: NullNode | null => None
208         case node: JsonNode => Some(node.asLong)
209       },

215         case _: NullNode | null => None
216         case node: JsonNode => Some(node.asLong)
217       }
218     )
219   }

231          |  `type`,                    -- String

259   def queryPartitionSpec(
260     database: String,
261     table: String
262   )(implicit nodeClient: NodeClient): Seq[PartitionSpec] = {
263     val partOutput = nodeClient.syncQueryAndCheckOutputJSONEachRow(
264       s"""SELECT
265          |  partition,                           -- String
266          |  partition_id,                        -- String
267          |  sum(rows)          AS row_count,     -- UInt64
268          |  sum(bytes_on_disk) AS size_in_bytes  -- UInt64
269          |FROM `system`.`parts`
270          |WHERE `database`='$database' AND `table`='$table' AND `active`=1
271          |GROUP BY `partition`, `partition_id`
272          |ORDER BY `partition` ASC, partition_id ASC
273          |""".stripMargin
274     )
275     if (partOutput.isEmpty || partOutput.rows == 1 && partOutput.records.head.get("partition").asText == "tuple()") {
276       return Array(NoPartitionSpec)
277     }
278     partOutput.records.map { row =>
279       PartitionSpec(
280         partition_value = row.get("partition").asText,
281         partition_id = row.get("partition_id").asText,
282         row_count = row.get("row_count").asLong,
283         size_in_bytes = row.get("size_in_bytes").asLong
284       )
285     }
286   }

298     database: String,
299     table: String,
300     partitionExpr: String,
301     cluster: Option[String] = None
302   )(implicit
303     nodeClient: NodeClient
304   ): Boolean =
305     nodeClient.syncQueryOutputJSONEachRow(
306       s"ALTER TABLE `$database`.`$table` ${cluster.map(c => s"ON CLUSTER $c").getOrElse("")} DROP PARTITION $partitionExpr"

308       case Right(_) => true
309       case Left(ex: CHException) =>
310         log.error(s"[${ex.code}]: ${ex.getMessage}")
