79   def stripSingleQuote(maybeQuoted: String): String = {
80     var start = 0
81     var until = maybeQuoted.length
82     if (maybeQuoted.startsWith("'")) start = 1
83     if (maybeQuoted.endsWith("'") && !maybeQuoted.endsWith("\\'")) until = until - 1
84     if (start > until) until = start
85     maybeQuoted.substring(start, until)
86   }

88   def wrapBackQuote(identifier: String): String = {
89     val sb = new mutable.StringBuilder(identifier.length + 2)
90     if (!identifier.startsWith("`")) sb.append('`')

92     if (identifier == "`" || !identifier.endsWith("`") || identifier.endsWith("\\`")) sb.append('`')
93     sb.mkString

96   @tailrec
97   def retry[R, T <: Throwable: ClassTag](retryTimes: Int, interval: Duration)(f: => R): Try[R] = {
98     assert(retryTimes >= 0)
99     val result = Try(f)
100     result match {
101       case Success(result) => Success(result)
102       case Failure(exception: T) if retryTimes > 0 =>
103         log.warn(s"Execution failed caused by: ", exception)
104         log.warn(s"$retryTimes times retry remaining, the next will be in ${interval.toMillis}ms")
105         LockSupport.parkNanos(interval.toNanos)
106         retry(retryTimes - 1, interval)(f)
107       case Failure(exception) => Failure(exception)
108     }
109   }

116   def bytesToString(size: BigInt): String = {
117     val EiB = 1L << 60
118     val PiB = 1L << 50
119     val TiB = 1L << 40
120     val GiB = 1L << 30
121     val MiB = 1L << 20
122     val KiB = 1L << 10
123 
124     if (size >= BigInt(1L << 11) * EiB) {
125       // The number is too large, show it in scientific notation.
126       BigDecimal(size, new MathContext(3, RoundingMode.HALF_UP)).toString() + " B"
127     } else {
128       val (value, unit) =
129         if (size >= 2 * EiB) {
130           (BigDecimal(size) / EiB, "EiB")
131         } else if (size >= 2 * PiB) {
132           (BigDecimal(size) / PiB, "PiB")
133         } else if (size >= 2 * TiB) {
134           (BigDecimal(size) / TiB, "TiB")
135         } else if (size >= 2 * GiB) {
136           (BigDecimal(size) / GiB, "GiB")
137         } else if (size >= 2 * MiB) {
138           (BigDecimal(size) / MiB, "MiB")
139         } else if (size >= 2 * KiB) {
140           (BigDecimal(size) / KiB, "KiB")
141         } else {
142           (BigDecimal(size), "B")
143         }
144       "%.1f %s".formatLocal(Locale.US, value, unit)
145     }
146   }

148   /**
149    * Returns a human-readable string representing a duration such as "35ms"
150    */
151   def msDurationToString(ms: Long): String = {
152     val second = 1000
153     val minute = 60 * second
154     val hour = 60 * minute
155     val locale = Locale.US
156 
157     ms match {
158       case t if t < second =>
159         "%d ms".formatLocal(locale, t)
160       case t if t < minute =>
161         "%.1f s".formatLocal(locale, t.toFloat / second)
162       case t if t < hour =>
163         "%.1f m".formatLocal(locale, t.toFloat / minute)
164       case t =>
165         "%.2f h".formatLocal(locale, t.toFloat / hour)
166     }
167   }
