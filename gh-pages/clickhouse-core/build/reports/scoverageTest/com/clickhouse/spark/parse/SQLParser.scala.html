33   private def parse[T](sql: String)(toResult: ClickHouseSQLParser => T): T = {
34 
35     log.debug(s"Parsing SQL: $sql")
36 
37     val lexer = new ClickHouseSQLLexer(CharStreams.fromString(sql))
38     lexer.removeErrorListeners()
39     lexer.addErrorListener(ParseErrorListener)
40 
41     val tokenStream = new CommonTokenStream(lexer)
42     val parser = new ClickHouseSQLParser(tokenStream)
43     parser.removeErrorListeners()
44     parser.addErrorListener(ParseErrorListener)
45 
46     try
47       try {
48         // first, try parsing with potentially faster SLL mode
49         parser.getInterpreter.setPredictionMode(PredictionMode.SLL)
50         toResult(parser)
51       } catch {
52         case _: ParseCancellationException =>
53           // if we fail, parse with LL mode
54           tokenStream.seek(0) // rewind input stream
55           parser.reset()
56 
57           // Try Again.
58           parser.getInterpreter.setPredictionMode(PredictionMode.LL)
59           toResult(parser)
60       }
61     catch {
62       case rethrow: ParseException => throw rethrow
63     }
64   }
65 }

67 object ParseErrorListener extends BaseErrorListener {
68   override def syntaxError(
69     recognizer: Recognizer[_, _],
70     offendingSymbol: scala.Any,
71     line: Int,
72     charPositionInLine: Int,
73     msg: String,
74     e: RecognitionException
75   ): Unit = {
76     val (start, stop) = offendingSymbol match {
77       case token: CommonToken =>
78         val start = Origin(Some(line), Some(token.getCharPositionInLine))
79         val length = token.getStopIndex - token.getStartIndex + 1
80         val stop = Origin(Some(line), Some(token.getCharPositionInLine + length))
81         (start, stop)
82       case _ =>
83         val start = Origin(Some(line), Some(charPositionInLine))
84         (start, start)
85     }
86     throw ParseException(msg, start, stop)
87   }
88 }
