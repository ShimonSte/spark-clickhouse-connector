45   private lazy val userAgent = {
46     val title = getClass.getPackage.getImplementationTitle
47     val version = getClass.getPackage.getImplementationVersion
48     if (version != null && title != null) {
49       val versions = version.split("_")
50       if (versions.length < 3) {
51         "Spark-ClickHouse-Connector"
52       } else {
53         val sparkVersion = versions(0)
54         val scalaVersion = versions(1)
55         val connectorVersion = versions(2)
56         s"${title}/${connectorVersion} (fv:spark/${sparkVersion}, lv:scala/${scalaVersion})"
57       }
58     } else {
59       "Spark-ClickHouse-Connector"
60     }
61   }

129   def syncInsert[OUT](
130     database: String,
131     table: String,
132     inputFormat: String,
133     inputCompressionType: ClickHouseCompression,
134     data: InputStream,
135     outputFormat: String,
136     deserializer: InputStream => SimpleOutput[OUT],
137     settings: Map[String, String]
138   ): Either[CHException, SimpleOutput[OUT]] = {
139     val queryId = nextQueryId()
140     val sql = s"INSERT INTO `$database`.`$table` FORMAT $inputFormat"
141     onExecuteQuery(queryId, sql)
142     val req = client.write(node)
143       .query(sql, queryId)
144       .decompressClientRequest(inputCompressionType)
145       .format(ClickHouseFormat.valueOf(outputFormat))
146     settings.foreach { case (k, v) => req.set(k, v) }
147     Try(req.data(data).executeAndWait()) match {
148       case Success(resp) => Right(deserializer(resp.getInputStream))
149       case Failure(ex: ClickHouseException) =>
150         Left(CHServerException(ex.getErrorCode, ex.getMessage, Some(nodeSpec), Some(ex)))
151       case Failure(ex) => Left(CHClientException(ex.getMessage, Some(nodeSpec), Some(ex)))
152     }
153   }
154 
155   def syncQuery[OUT](
156     sql: String,
157     outputFormat: String,
158     deserializer: InputStream => SimpleOutput[OUT],
159     settings: Map[String, String]
160   ): Either[CHException, SimpleOutput[OUT]] = {
161     val queryId = nextQueryId()
162     onExecuteQuery(queryId, sql)
163     val req = client.read(node)
164       .query(sql, queryId).asInstanceOf[ClickHouseRequest[_]]
165       .format(ClickHouseFormat.valueOf(outputFormat)).asInstanceOf[ClickHouseRequest[_]]
166       .option(ClickHouseClientOption.CONNECTION_TIMEOUT, timeout).asInstanceOf[ClickHouseRequest[_]]
167     settings.foreach { case (k, v) => req.set(k, v).asInstanceOf[ClickHouseRequest[_]] }
168     Try(req.executeAndWait()) match {
169       case Success(resp) => Right(deserializer(resp.getInputStream))
170       case Failure(ex: ClickHouseException) =>
171         Left(CHServerException(ex.getErrorCode, ex.getMessage, Some(nodeSpec), Some(ex)))
172       case Failure(ex) => Left(CHClientException(ex.getMessage, Some(nodeSpec), Some(ex)))
173     }
174   }

178     outputFormat: String,

182     case Left(rethrow) => throw rethrow

184   }

194     settings: Map[String, String] = Map.empty
195   ): ClickHouseResponse = {
196     val queryId = nextQueryId()

200       .compressServerResponse(outputCompressionType).asInstanceOf[ClickHouseRequest[_]]
201       .format(ClickHouseFormat.valueOf(outputFormat)).asInstanceOf[ClickHouseRequest[_]]
202       .option(ClickHouseClientOption.CONNECTION_TIMEOUT, timeout).asInstanceOf[ClickHouseRequest[_]]
203     settings.foreach { case (k, v) => req.set(k, v).asInstanceOf[ClickHouseRequest[_]] }
204     Try(req.executeAndWait()) match {
205       case Success(resp) => resp
206       case Failure(ex: ClickHouseException) =>
207         throw CHServerException(ex.getErrorCode, ex.getMessage, Some(nodeSpec), Some(ex))
208       case Failure(ex) => throw CHClientException(ex.getMessage, Some(nodeSpec), Some(ex))
209     }
210   }
