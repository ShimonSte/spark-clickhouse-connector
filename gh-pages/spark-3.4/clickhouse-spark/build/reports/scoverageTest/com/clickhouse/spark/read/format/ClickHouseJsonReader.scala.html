44   override def decode(record: Array[JsonNode]): InternalRow = {
45     val values: Array[Any] = new Array[Any](record.length)
46     if (readSchema.nonEmpty) {
47       var i: Int = 0
48       while (i < record.length) {
49         values(i) = decodeValue(record(i), readSchema.fields(i))
50         i = i + 1
51       }
52     }
53     new GenericInternalRow(values)
54   }
55 
56   private def decodeValue(jsonNode: JsonNode, structField: StructField): Any = {
57     if (jsonNode == null || jsonNode.isNull) {
58       // should we check `structField.nullable`?
59       return null
60     }
61 
62     structField.dataType match {
63       case BooleanType => jsonNode.asBoolean
64       case ByteType => jsonNode.asInt.byteValue
65       case ShortType => jsonNode.asInt.shortValue
66       case IntegerType => jsonNode.asInt
67       case LongType => jsonNode.asLong
68       case FloatType => jsonNode.asDouble.floatValue
69       case DoubleType => jsonNode.asDouble
70       case d: DecimalType if jsonNode.isBigDecimal =>
71         Decimal(jsonNode.decimalValue, d.precision, d.scale)
72       case d: DecimalType if jsonNode.isFloat | jsonNode.isDouble =>
73         Decimal(BigDecimal(jsonNode.doubleValue, new MathContext(d.precision)), d.precision, d.scale)
74       case d: DecimalType if jsonNode.isInt =>
75         Decimal(BigDecimal(jsonNode.intValue, new MathContext(d.precision)), d.precision, d.scale)
76       case d: DecimalType if jsonNode.isLong =>
77         Decimal(BigDecimal(jsonNode.longValue, new MathContext(d.precision)), d.precision, d.scale)
78       case d: DecimalType if jsonNode.isBigInteger =>
79         Decimal(BigDecimal(jsonNode.bigIntegerValue, new MathContext(d.precision)), d.precision, d.scale)
80       case d: DecimalType =>
81         Decimal(BigDecimal(jsonNode.textValue, new MathContext(d.precision)), d.precision, d.scale)
82       case TimestampType =>
83         var _instant =
84           ZonedDateTime.parse(jsonNode.asText, dateTimeFmt.withZone(scanJob.tz)).withZoneSameInstant(ZoneOffset.UTC)
85         TimeUnit.SECONDS.toMicros(_instant.toEpochSecond) + TimeUnit.NANOSECONDS.toMicros(_instant.getNano())
86       case StringType => UTF8String.fromString(jsonNode.asText)
87       case DateType => LocalDate.parse(jsonNode.asText, dateFmt).toEpochDay.toInt
88       case BinaryType => jsonNode.binaryValue
89       case ArrayType(_dataType, _nullable) =>
90         val _structField = StructField(s"${structField.name}__array_element__", _dataType, _nullable)
91         new GenericArrayData(jsonNode.asScala.map(decodeValue(_, _structField)))
92       case MapType(StringType, _valueType, _valueNullable) =>
93         val mapData = jsonNode.fields.asScala.map { entry =>
94           val _structField = StructField(s"${structField.name}__map_value__", _valueType, _valueNullable)
95           UTF8String.fromString(entry.getKey) -> decodeValue(entry.getValue, _structField)
96         }.toMap
97         ArrayBasedMapData(mapData)
98       case _ =>
99         throw CHClientException(s"Unsupported catalyst type ${structField.name}[${structField.dataType}]")
100     }
101   }
