122           case Left(exception) if exception.code == UNKNOWN_TABLE.code =>
123             throw new NoSuchTableException(ident)

151    * <h2>MergeTree Engine</h2>
152    * {{{
153    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
154    * (
155    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1] [TTL expr1],
156    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2] [TTL expr2],
157    *     ...
158    *     INDEX index_name1 expr1 TYPE type1(...) GRANULARITY value1,
159    *     INDEX index_name2 expr2 TYPE type2(...) GRANULARITY value2
160    * ) ENGINE = MergeTree()
161    * ORDER BY expr
162    * [PARTITION BY expr]
163    * [PRIMARY KEY expr]
164    * [SAMPLE BY expr]
165    * [TTL expr
166    *     [DELETE|TO DISK 'xxx'|TO VOLUME 'xxx' [, ...] ]
167    *     [WHERE conditions]
168    *     [GROUP BY key_expr [SET v1 = agg_func(v1) [, v2 = agg_func(v2) ...]] ]]
169    * [SETTINGS name=value, ...]
170    * }}}
171    * <p>
172    *
173    * <h2>ReplacingMergeTree Engine</h2>
174    * {{{
175    * CREATE TABLE [IF NOT EXISTS] [db.]table_name [ON CLUSTER cluster]
176    * (
177    *     name1 [type1] [DEFAULT|MATERIALIZED|ALIAS expr1],
178    *     name2 [type2] [DEFAULT|MATERIALIZED|ALIAS expr2],
179    *     ...
180    * ) ENGINE = ReplacingMergeTree([ver])
181    * [PARTITION BY expr]
182    * [ORDER BY expr]
183    * [PRIMARY KEY expr]
184    * [SAMPLE BY expr]
185    * [SETTINGS name=value, ...]
186    * }}}
187    *
188    * `ver` — column with version. Type `UInt*`, `Date` or `DateTime`.

192   override def createTable(
193     ident: Identifier,
194     schema: StructType,
195     partitions: Array[Transform],
196     properties: util.Map[String, String]
197   ): ClickHouseTable = {
198     val (db, tbl) = unwrap(ident) match {
199       case Some((d, t)) => (d, t)
200       case None => throw CHClientException(s"Invalid table identifier: $ident")
201     }
202     val props = properties.asScala
203 
204     val engineExpr = props.getOrElse("engine", "MergeTree()")
205 
206     val isCreatingDistributed = engineExpr equalsIgnoreCase "Distributed"
207     val keyPrefix = if (isCreatingDistributed) "local." else ""
208 
209     val partitionsClause = partitions match {
210       case transforms if transforms.nonEmpty =>
211         transforms.map(ExprUtils.toClickHouse(_, functionRegistry).sql).mkString("PARTITION BY (", ", ", ")")
212       case _ => ""
213     }
214 
215     val orderClause = props.get(s"${keyPrefix}order_by").map(o => s"ORDER BY ($o)").getOrElse("")
216     val primaryKeyClause = props.get(s"${keyPrefix}primary_key").map(p => s"PRIMARY KEY ($p)").getOrElse("")
217     val sampleClause = props.get(s"${keyPrefix}sample_by").map(p => s"SAMPLE BY ($p)").getOrElse("")
218 
219     val fieldsClause = SchemaUtils
220       .toClickHouseSchema(schema)
221       .map { case (fieldName, ckType, comment) => s"${quoted(fieldName)} $ckType$comment" }
222       .mkString(",\n ")
223 
224     val clusterOpt = props.get("cluster")
225 
226     def tblSettingsClause(prefix: String): String = props.filterKeys(_.startsWith(prefix)) match {
227       case settings if settings.nonEmpty =>
228         settings.map { case (k, v) =>
229           s"${k.substring(prefix.length)}=$v"
230         }.mkString("SETTINGS ", ", ", "")
231       case _ => ""
232     }
233 
234     def createTable(
235       clusterOpt: Option[String],
236       engineExpr: String,
237       database: String,
238       table: String,
239       settingsClause: String
240     ): Unit = {
241       val clusterClause = clusterOpt.map(c => s"ON CLUSTER $c").getOrElse("")
242       nodeClient.syncQueryAndCheckOutputJSONEachRow(
243         s"""CREATE TABLE `$database`.`$table` $clusterClause (
244            |$fieldsClause
245            |) ENGINE = $engineExpr
246            |$partitionsClause
247            |$orderClause
248            |$primaryKeyClause
249            |$sampleClause
250            |$settingsClause
251            |""".stripMargin
252           .replaceAll("""\n\s+\n""", "\n") // remove empty lines
253       )
254     }
255 
256     def createDistributedTable(
257       cluster: String,
258       shardExpr: String,
259       localDatabase: String,
260       localTable: String,
261       distributedDatabase: String,
262       distributedTable: String,
263       settingsClause: String
264     ): Unit = nodeClient.syncQueryAndCheckOutputJSONEachRow(
265       s"""CREATE TABLE `$distributedDatabase`.`$distributedTable` ON CLUSTER $cluster
266          |AS `$localDatabase`.`$localTable`
267          |ENGINE = Distributed($cluster, '$localDatabase', '$localTable', ($shardExpr))
268          |$settingsClause
269          |""".stripMargin
270     )
271 
272     if (isCreatingDistributed) {
273       val cluster = clusterOpt.getOrElse("default")
274       val shardExpr = props.getOrElse("shard_by", "rand()")
275       val settingsClause = tblSettingsClause("settings.")
276       val localEngineExpr = props.getOrElse(s"${keyPrefix}engine", s"MergeTree()")
277       val localDatabase = props.getOrElse(s"${keyPrefix}database", db)
278       val localTable = props.getOrElse(s"${keyPrefix}table", s"${tbl}_local")
279       val localSettingsClause = tblSettingsClause(s"${keyPrefix}settings.")
280       createTable(Some(cluster), localEngineExpr, localDatabase, localTable, localSettingsClause)
281       createDistributedTable(cluster, shardExpr, localDatabase, localTable, db, tbl, settingsClause)
282     } else {
283       val settingsClause = tblSettingsClause(s"${keyPrefix}settings.")
284       createTable(clusterOpt, engineExpr, db, tbl, settingsClause)
285     }
286 
287     loadTable(ident)
288   }

294   override def dropTable(ident: Identifier): Boolean = {
295     val tableOpt =
296       try Some(loadTable(ident))
297       catch {
298         case _: NoSuchTableException => None
299       }
300     tableOpt match {
301       case None => false
302       case Some(ClickHouseTable(_, cluster, _, tableSpec, _, _)) =>
303         val (db, tbl) = (tableSpec.database, tableSpec.name)
304         val isAtomic = loadNamespaceMetadata(Array(db)).get("engine").equalsIgnoreCase("atomic")
305         val syncClause = if (isAtomic) "SYNC" else ""
306         // limitation: only support Distribute table, can not handle cases such as drop local table on cluster nodes
307         val clusterClause = cluster.map(c => s"ON CLUSTER ${c.name}").getOrElse("")
308         nodeClient.syncQueryOutputJSONEachRow(s"DROP TABLE `$db`.`$tbl` $clusterClause $syncClause").isRight
309     }
