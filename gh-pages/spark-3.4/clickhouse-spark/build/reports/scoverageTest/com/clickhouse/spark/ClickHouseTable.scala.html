188   override def dropPartition(ident: InternalRow): Boolean = {
189     val partitionExpr = (0 until ident.numFields).map { i =>
190       partitionSchema.fields(i).dataType match {
191         case IntegerType => compileValue(ident.getInt(i))
192         case LongType => compileValue(ident.getLong(i))
193         case StringType => compileValue(ident.getUTF8String(i))
194         case DateType => compileValue(LocalDate.ofEpochDay(ident.getInt(i)))
195         case illegal => throw new IllegalArgumentException(s"Illegal partition data type: $illegal")
196       }
197     }.mkString("(", ",", ")")
198 
199     Utils.tryWithResource(NodeClient(node)) { implicit nodeClient =>
200       engineSpec match {
201         case DistributedEngineSpec(_, cluster, local_db, local_table, _, _) =>
202           dropPartition(local_db, local_table, partitionExpr, Some(cluster))
203         case _ =>
204           dropPartition(database, table, partitionExpr)
205       }
206     }
207   }

219   override def listPartitionIdentifiers(names: Array[String], ident: InternalRow): Array[InternalRow] = {
220     assert(
221       names.length == ident.numFields,
222       s"Number of partition names (${names.length}) must be equal to " +
223         s"the number of partition values (${ident.numFields})."
224     )
225     assert(
226       names.forall(fieldName => partitionSchema.fieldNames.contains(fieldName)),
227       s"Some partition names ${names.mkString("[", ", ", "]")} don't belong to " +
228         s"the partition schema '${partitionSchema.sql}'."
229     )

231     def strToSparkValue(str: String, dataType: DataType): Any = dataType match {
232       case StringType => UTF8String.fromString(str.stripPrefix("'").stripSuffix("'"))
233       case IntegerType => JInt.parseInt(str)
234       case LongType => JLong.parseLong(str)
235       case DateType => LocalDate.parse(str.stripPrefix("'").stripSuffix("'"), dateFmt).toEpochDay.toInt
236       case unsupported => throw new UnsupportedOperationException(s"$unsupported")
237     }
238 
239     val partitionSpecs: Seq[PartitionSpec] = engineSpec match {
240       case DistributedEngineSpec(_, _, local_db, local_table, _, _) =>
241         cluster.get.shards.flatMap { shardSpec =>
242           Utils.tryWithResource(NodeClient(shardSpec.nodes.head)) { implicit nodeClient: NodeClient =>
243             queryPartitionSpec(local_db, local_table)
244           }
245         }
246       case _ =>
247         Utils.tryWithResource(NodeClient(node)) { implicit nodeClient =>
248           queryPartitionSpec(database, table)
249         }
250     }
251     partitionSpecs.map(_.partition_value)
252       .distinct
253       .filterNot(_.isEmpty) // represent partitioned table w/o records
254       .filterNot(_ == "tuple()") // represent the root partition of un-partitioned table
255       .map {
256         case tuple if tuple.startsWith("(") && tuple.endsWith(")") =>
257           tuple.stripPrefix("(").stripSuffix(")").split(",")
258         case partColStrValue =>
259           Array(partColStrValue)
260       }
261       .map { partColStrValues =>
262         new GenericInternalRow(
263           (partColStrValues zip partitionSchema.fields.map(_.dataType))
264             .map { case (partColStrValue, dataType) => strToSparkValue(partColStrValue, dataType) }
265         )
266       }
267       .filter { partRow =>
268         names.zipWithIndex.forall { case (name, queryIndex) =>
269           val partRowIndex = partitionSchema.fieldIndex(name)
270           val dataType = partitionSchema.fields(partRowIndex).dataType
271           partRow.get(partRowIndex, dataType) == ident.get(queryIndex, dataType)
272         }
273       }
274       .toArray
275   }
